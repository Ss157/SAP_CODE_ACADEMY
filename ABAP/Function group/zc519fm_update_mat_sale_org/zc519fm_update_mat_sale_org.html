FUNCTION zc519fm_update_mat_sale_org.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IS_MAT) LIKE  ZC519T0022 STRUCTURE  ZC519T0022
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE erdat, erzet, ernam
    INTO @DATA(ls_data)
    FROM ZC519t0022
   WHERE matnr = @is_mat-matnr
     AND auart = @is_mat-auart.


  is_mat = VALUE #( BASE is_mat
                    erdat = ls_data-erdat
                    erzet = ls_data-erzet
                    ernam = ls_data-ernam

                    aedat = sy-datum
                    aezet = sy-uzeit
                    aenam = sy-uname
                  ).

  UPDATE ZC519t0022 FROM is_mat.

  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.
  ELSE.
    ROLLBACK WORK.
  ENDIF.

ENDFUNCTION.
