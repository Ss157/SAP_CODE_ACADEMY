FUNCTION zc519fm_get_purchase2.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  EXPORTING
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  TABLES
*"      ET_PURCHASE STRUCTURE  ZC519S0013
*"      IT_EBELN OPTIONAL
*"----------------------------------------------------------------------

*       Global data declarations

  CLEAR   : ev_code, ev_msg.
  REFRESH : et_purchase.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~ktmng b~meins
    INTO CORRESPONDING FIELDS OF TABLE et_purchase
    FROM ekko      AS a
   INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
   WHERE a~ebeln IN it_ebeln.

  IF sy-subrc <> 0.

    ev_code = 'E'.
    ev_msg  = 'Data not found'.

  ELSE.

    ev_code = 'S'.

  ENDIF.

ENDFUNCTION.
