FUNCTION zc519fm_get_mat_sale_org_list.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR OPTIONAL
*"     VALUE(IV_AUART) TYPE  AUART OPTIONAL
*"  TABLES
*"      ET_LIST STRUCTURE  ZC519T0022
*"----------------------------------------------------------------------

*       Global data declarations

  RANGES : lr_matnr FOR ZC519t0022-matnr,
           lr_auart FOR ZC519t0022-auart.

  lr_matnr[] = COND #( WHEN iv_matnr NE space
                        THEN VALUE #(
                                      (
                                        sign   = 'I'
                                        option = 'EQ'
                                        low    = iv_matnr
                                      )
                                     )
                     ).

  lr_auart[] = COND #( WHEN iv_auart NE space
                        THEN VALUE #(
                                      (
                                        sign   = 'I'
                                        option = 'EQ'
                                        low    = iv_auart
                                      )
                                     )
                     ).

  SELECT matnr auart vkorg spart
    INTO CORRESPONDING FIELDS OF TABLE et_list
    FROM zc519t0022
   WHERE matnr IN lr_matnr
     AND auart IN lr_auart.

ENDFUNCTION.
