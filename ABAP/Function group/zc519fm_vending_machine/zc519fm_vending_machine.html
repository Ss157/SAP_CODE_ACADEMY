FUNCTION zc519fm_vending_machine.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_CASH) TYPE  INT4
*"  EXPORTING
*"     REFERENCE(PE_NAME) TYPE  CHAR20
*"----------------------------------------------------------------------

*       Global data declarations

  CASE pi_cash.
    WHEN 1500.
      pe_name = '레쓰비'.

    WHEN 1800.
      pe_name = 'TOP'.

    WHEN 2000.
      pe_name = '메가커피'.

    WHEN 3000.

    WHEN 5000.
      pe_name = 'TERRA캔'.

    WHEN OTHERS.
      pe_name = '돈먹음'.

  ENDCASE.

ENDFUNCTION.
