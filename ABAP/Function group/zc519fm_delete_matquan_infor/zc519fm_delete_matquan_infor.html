FUNCTION zc519fm_delete_matquan_infor.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_WERKS) TYPE  WERKS_D
*"  EXCEPTIONS
*"      DELETE_FAIL
*"----------------------------------------------------------------------

*       Global data declarations

  DELETE FROM ZC519t0018
   WHERE matnr = iv_matnr
     AND werks = iv_werks.

  IF sy-subrc = 0.
    COMMIT WORK AND WAIT.

    ROLLBACK WORK.

  ENDIF.

ENDFUNCTION.
