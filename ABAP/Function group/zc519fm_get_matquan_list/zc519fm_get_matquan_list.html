Description: [C5] Material Quantity List 
FUNCTION zc519fm_get_matquan_list.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR OPTIONAL
*"     VALUE(IV_WERKS) TYPE  WERKS_D OPTIONAL
*"  TABLES
*"      ET_LIST STRUCTURE  ZC519T0018
*"----------------------------------------------------------------------

*       Global data declarations

  RANGES : lr_matnr FOR ZC519t0018-matnr,
           lr_werks FOR ZC519t0018-werks.

  IF iv_matnr IS NOT INITIAL.

    lr_matnr[] = VALUE #(
                          (
                            sign   = 'I'
                            option = 'EQ'
                            low    = iv_matnr
                          )
                        ).

  ENDIF.

  IF iv_werks IS NOT INITIAL.

    lr_werks[] = VALUE #(
                          (
                            sign   = 'I'
                            option = 'EQ'
                            low    = iv_werks
                          )
                        ).

  ENDIF.

  SELECT matnr werks mtart menge meins dmbtr waers
    INTO CORRESPONDING FIELDS OF TABLE et_list
    FROM zc519t0018
   WHERE matnr IN lr_matnr
     AND werks IN lr_werks.

ENDFUNCTION.
