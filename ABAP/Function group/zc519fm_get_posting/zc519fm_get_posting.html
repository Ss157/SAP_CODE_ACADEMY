FUNCTION zc519fm_get_posting.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_BUKRS) TYPE  BUKRS
*"     REFERENCE(IV_GJAHR) TYPE  GJAHR
*"  EXPORTING
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  TABLES
*"      ET_DATA STRUCTURE  ZC519S0011
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT a~blart a~budat
           b~belnr b~buzei b~bschl b~koart
           b~gsber b~shkzg b~dmbtr b~pswsl
      INTO CORRESPONDING FIELDS OF TABLE et_data
      FROM bkpf      AS a
     INNER JOIN bseg AS b
        ON a~belnr EQ b~belnr
       AND a~bukrs EQ b~bukrs
       AND a~gjahr EQ b~gjahr
     WHERE a~bukrs EQ iv_bukrs
       AND a~gjahr EQ iv_gjahr.

  IF sy-subrc NE 0.

    ev_code = 'E'.
    ev_msg  = 'Data not found'.

  ELSE.

    ev_code = 'S'.

  ENDIF.

ENDFUNCTION.
