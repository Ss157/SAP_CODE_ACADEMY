Description: [C5] Get Material Quantity and Amount 
FUNCTION zc519fm_get_matquan_infor.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_WERKS) TYPE  WERKS_D
*"  EXPORTING
*"     VALUE(ES_MAT) TYPE  ZC519T0018
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE matnr werks mtart
         menge  meins dmbtr waers
         erdat  erzet ernam aedat
         aezet  aenam
    INTO CORRESPONDING FIELDS OF es_mat
    FROM zc519t0018
   WHERE matnr = iv_matnr
     AND werks = iv_werks.

ENDFUNCTION.
