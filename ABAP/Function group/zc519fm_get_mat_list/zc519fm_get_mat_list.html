FUNCTION zc519fm_get_mat_list.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR OPTIONAL
*"     VALUE(IV_WERKS) TYPE  WERKS_D OPTIONAL
*"     VALUE(IV_MTART) TYPE  MTART OPTIONAL
*"  TABLES
*"      ET_LIST STRUCTURE  ZC519T0017
*"----------------------------------------------------------------------

*       Global data declarations

  RANGES : lr_matnr FOR ZC519t0017-matnr,
           lr_werks FOR ZC519t0017-werks,
           lr_mtart FOR ZC519t0017-mtart.

*  IF lr_matnr IS NOT INITIAL.
** ranges는 헤더 있는 인터널 테이블이기 때문에
** []를 사용해서 바디에 넣어야 한다.
*    lr_matnr[] = VALUE #(
*                          (
*                            sign   = 'I'
*                            option = 'EQ'
*                            low    = iv_matnr
*                          )
*                        ).
*
*  ENDIF.

  lr_matnr[] = COND #( WHEN iv_matnr NE space
                        THEN VALUE #(
                                      (
                                        sign   = 'I'
                                        option = 'EQ'
                                        low    = iv_matnr
                                      )
                                     )
                     ).

*  IF lr_werks IS NOT INITIAL.
** ranges는 헤더 있는 인터널 테이블이기 때문에
** []를 사용해서 바디에 넣어야 한다.
*    lr_werks[] = VALUE #(
*                          (
*                            sign   = 'I'
*                            option = 'EQ'
*                            low    = iv_werks
*                          )
*                        ).
*
*  ENDIF.

  lr_werks[] = COND #( WHEN iv_werks NE space
                        THEN VALUE #(
                                      (
                                        sign   = 'I'
                                        option = 'EQ'
                                        low    = iv_werks
                                      )
                                     )
                     ).

*  IF lr_mtart IS NOT INITIAL.
** ranges는 헤더 있는 인터널 테이블이기 때문에
** []를 사용해서 바디에 넣어야 한다.
*    lr_mtart[] = VALUE #(
*                          (
*                            sign   = 'I'
*                            option = 'EQ'
*                            low    = iv_mtart
*                          )
*                        ).
*
*  ENDIF.

  lr_mtart[] = COND #( WHEN iv_mtart NE space
                        THEN VALUE #(
                                      (
                                        sign   = 'I'
                                        option = 'EQ'
                                        low    = iv_mtart
                                      )
                                     )
                     ).

  SELECT matnr werks lgort bwart mtart matkl dispo
    INTO CORRESPONDING FIELDS OF TABLE et_list
    FROM ZC519t0017
   WHERE matnr IN lr_matnr
     AND werks IN lr_werks
     AND mtart IN lr_mtart.

ENDFUNCTION.
