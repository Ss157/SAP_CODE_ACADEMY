FUNCTION zc519fm_get_maktx.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_MATNR) TYPE  MATNR
*"     REFERENCE(IV_SPRAS) TYPE  SPRAS OPTIONAL
*"  EXPORTING
*"     REFERENCE(EV_MAKTX) TYPE  MAKTX
*"     REFERENCE(EV_CODE) TYPE  C
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  EXCEPTIONS
*"      MATNR_REQUIRED
*"      DATA_NOT_FOUND
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lv_spras TYPE sy-langu.

* Export Parameter를 사용한 경우
  IF iv_matnr IS INITIAL.

    ev_code = 'E'.
    ev_msg = 'Material code is required'.
    EXIT.

  ENDIF.

* Exception을 사용한 경우
*  IF iv_matnr IS INITIAL.
*
*    RAISE matnr_required. " EXIT이 필요없다 exception이 발생하면 자동으로 종료된다.
*
*  ENDIF.

* iv_spras : optional인 파라미터는 입력이 안되더라도 처리를 해줘야 함.
  IF iv_spras IS INITIAL.

    lv_spras = sy-langu.

  ELSE.

    lv_spras = iv_spras.

  ENDIF.

  SELECT SINGLE maktx
    INTO ev_maktx
    FROM makt
   WHERE spras = lv_spras
     AND matnr = iv_matnr.

* Export Parameter 사용한 경우
  IF sy-subrc <> 0.

    ev_code = 'E'.
    ev_msg  = 'Data not found'.

  ELSE.

    ev_code = 'S'.

  ENDIF.

ENDFUNCTION.
