FUNCTION zc519fm_get_airline_info.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_CARRID) TYPE  S_CARR_ID
*"  EXPORTING
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  TABLES
*"      ET_DATA STRUCTURE  SFLIGHT
*"----------------------------------------------------------------------

*       Global data declarations

  CLEAR : et_data, ev_code, ev_msg.

  SELECT carrid     connid     fldate     price
         currency   planetype  seatsmax   seatsocc
         paymentsum seatsmax_b seatsocc_b seatsmax_f seatsocc_f
    INTO CORRESPONDING FIELDS OF TABLE et_data
    FROM sflight
   WHERE carrid = iv_carrid.

  IF sy-subrc <> 0.

    ev_code = 'E'.
    ev_msg  = 'Data not found'.

  ELSE.

    ev_code = 'S'.

  ENDIF.

ENDFUNCTION.
