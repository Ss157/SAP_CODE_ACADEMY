Description: [C5] Get Material Information 
FUNCTION zc519fm_get_mat_infor.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_WERKS) TYPE  WERKS_D
*"  EXPORTING
*"     VALUE(ES_MAT) TYPE  ZC519T0017
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE matnr werks lgort
         bwart  mtart matkl dispo
    INTO CORRESPONDING FIELDS OF es_mat
    FROM ZC519t0017
   WHERE matnr = iv_matnr
     AND werks = iv_werks.


ENDFUNCTION.
