FUNCTION zc519fm_get_mat_sale_org.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     VALUE(IV_MATNR) TYPE  MATNR
*"     VALUE(IV_AUART) TYPE  AUART
*"  EXPORTING
*"     VALUE(ES_MAT) TYPE  ZC519T0022
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE matnr auart vkorg spart
         erdat  erzet ernam
         aedat  aezet aenam
    INTO CORRESPONDING FIELDS OF es_mat
    FROM zc519t0022
   WHERE matnr = iv_matnr
     AND auart = iv_auart.

ENDFUNCTION.
