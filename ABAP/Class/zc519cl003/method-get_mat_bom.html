  METHOD get_mat_bom.

    CLEAR   : ev_code, ev_msg.
    REFRESH : et_data.

    SELECT a~matnr a~werks a~stlan
           b~stlnr b~stlty
           c~stlkn c~idnrk c~postp c~posnr
           c~menge c~meins c~sanka
           d~maktx
           e~maktx AS idnrk_txt
      INTO CORRESPONDING FIELDS OF TABLE et_data
      FROM mast      AS a
     INNER JOIN stko AS b
        ON a~stlnr = b~stlnr
       AND a~stlal = b~stlal
     INNER JOIN stpo AS c
        ON b~stlty = c~stlty
       AND b~stlnr = c~stlnr
      LEFT OUTER JOIN makt AS d
        ON a~matnr = d~matnr
       AND d~spras = sy-langu
      LEFT OUTER JOIN makt AS e
        ON c~idnrk = e~matnr
       AND e~spras = sy-langu
     WHERE a~werks  = iv_werks
       AND a~matnr IN it_matnr.

    IF sy-subrc <> 0.

      ev_code = 'E'.
      ev_msg  = 'Data not found'.

    ELSE.

      ev_code = 'S'.

    ENDIF.

  ENDMETHOD.
