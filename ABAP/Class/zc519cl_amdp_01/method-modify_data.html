  METHOD modify_data bY DATABASE PROCEDURE FOR HDB
                     LANGUAGE SQLSCRIPT
                     USING scarr.

    DECLARE lv_tabix iNTEGER := 0;
    DECLARE aCarrid   NVARCHAR( 3 )   ARRAY;
    DECLARE aCarrname NVARCHAR( 20 )  ARRAY;
    DECLARE aCurrcode NVARCHAR( 5 )   ARRAY;
    DECLARE aUrl      NVARCHAR( 100 ) ARRAY;
*lt_scarr는 SELECT 문을 받을 테이블이라고 생각하면 됨.
    DECLARE CURSOR lt_scarr FOR
            SELECT carrid, carrname, currcode, url
              FROM scarr
             WHERE mandt = session_context( 'CLIENT' );
*    LOOP 문 돌리는 중
    FOR ls_scarr AS lt_scarr DO
      lv_tabix := lv_tabix + 1; --INDEX 하나씩 증가

      aCarrid[lv_tabix]   := ls_scarr.carrid;
      aCarrname[lv_tabix] := ls_scarr.carrname;
      aCurrcode[lv_tabix] := ls_scarr.currcode;
      aUrl[lv_tabix]      := ls_scarr.url;

      IF ls_scarr.carrid = 'JL'
      THEN aCarrid[lv_tabix]   := 'KA';
           aCarrname[lv_tabix] := 'Korean Air';
           aCurrcode[lv_tabix] := 'KRW';
           aUrl[lv_tabix]      := 'www.korean-air.co.kr';
      END IF;

    END FOR;

    et_list = UNNEST (:aCarrid, :aCarrname, :aCurrcode, :aUrl)
              AS ("CARRID", "CARRNAME", "CURRCODE", "URL");

  ENDMETHOD.
