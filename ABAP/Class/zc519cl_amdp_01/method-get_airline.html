  METHOD get_airline BY DATABASE PROCEDURE FOR HDB
                     LANGUAGE SQLSCRIPT
                     USING scarr.

    -- ABAP 코드가 아니니까 INTO TABLE이 안 먹힌다.
    -- 그래서 et_airline = 로 레코드를 받는 것
    -- et_airline 과 SELECT 하는 필드의 개수는 반드시 동일해야 함
    -- 신문법 @ --> AMDP :

            et_airline = SELECT carrid, carrname, currcode, url
                           FROM scarr
                          WHERE mandt  = session_context( 'CLIENT' )
                            AND carrid = :iv_carrid;

  ENDMETHOD.
