  METHOD get_fi_doc BY DATABASE PROCEDURE FOR HDB
                    LANGUAGE SQLSCRIPT
                    USING bkpf zc519cl_amdp_01=>get_bseg.

    et_fi_doc = select a.bukrs, a.belnr, b.buzei,
                       b.gjahr, b.hkont, b.wrbtr, b.pswsl
                  fROM bkpf as a
                 inner join "ZC519CL_AMDP_01=>GET_BSEG"() AS b
                    on a.mandt = b.mandt
                   and a.bukrs = b.bukrs
                   and a.belnr = b.belnr
                 where a.mandt = session_context( 'CLIENT' )
                   and a.bukrs = :iv_bukrs
                   and a.gjahr = :iv_gjahr;

  ENDMETHOD.
