  METHOD get_order.

    CLEAR   : ev_code, ev_msg.
    REFRESH : et_order.

    SELECT a~aufnr a~gstrp a~gltrp a~plnbez a~gamng a~gmein
           b~posnr b~matnr b~psmng b~amein  b~bwtar
           c~maktx AS plnbez_txt
           d~maktx
      INTO CORRESPONDING FIELDS OF TABLE et_order
      FROM afko      AS a
     INNER JOIN afpo AS b
        ON a~aufnr = b~aufnr
      LEFT OUTER JOIN makt AS c
        ON a~plnbez = c~matnr
       AND c~spras  = sy-langu
      LEFT OUTER JOIN makt AS d
        ON b~matnr = d~matnr
       AND d~spras = sy-langu
     WHERE a~plnbez IN it_plnbez
       AND b~pwerk   = iv_pwerk.

    IF sy-subrc <> 0.

      ev_code = 'E'.
      ev_msg  = 'Data not found'.

    ELSE.

      ev_code = 'S'.

    ENDIF.

  ENDMETHOD.
