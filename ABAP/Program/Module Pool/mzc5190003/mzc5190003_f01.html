*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'MENGE'.
      gs_fcat-qfieldname = 'MEINS'.

    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'WAERS'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_data .

  DATA : lv_cnt TYPE i.

  CLEAR  lv_cnt.
  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  lv_cnt = lines( gt_row ).
*  DESCRIBE TABLE gt_row LINES lv_cnt.

  IF lv_cnt <> 1.

    MESSAGE s000 WITH TEXT-f02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_mat INTO gs_mat INDEX gs_row-index.

ENDFORM.

FORM get_mat .

  _clear gs_mat gt_mat.

  SELECT matnr werks mtart menge meins dmbtr waers
         erdat erzet ernam aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM zc519t0018
   ORDER BY matnr werks.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc519t0018.

  CLEAR ls_save.

  IF gs_mat-matnr IS INITIAL OR
     gs_mat-werks IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'. "Material and Plant is required field
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_mat TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc519t0018 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_mat.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM delete_data .

  IF gs_mat-matnr IS INITIAL OR
     gs_mat-werks IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc519t0018
   WHERE matnr = gs_mat-matnr
     AND werks = gs_mat-werks.

  IF sy-subrc = 0.

    COMMIT WORK AND WAIT.

    PERFORM get_mat.
    PERFORM refresh_grid.

    MESSAGE s003.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM f4_mtart .

  DATA : BEGIN OF ls_mtart,
*           INCLUDE STRUCTURE zc519t0018.
           mtart TYPE t134-mtart,
           vmtpo TYPE t134-vmtpo,
           vprsv TYPE t134-vprsv,
           mtbez TYPE t134t-mtbez,
         END OF ls_mtart,

         lt_mtart LIKE TABLE OF ls_mtart.

  _clear ls_mtart lt_mtart.

  SELECT a~mtart a~vmtpo a~vprsv
         b~mtbez
    INTO CORRESPONDING FIELDS OF TABLE lt_mtart
    FROM t134             AS a
    LEFT OUTER JOIN t134t AS b
      ON a~mtart = b~mtart
     AND b~spras = sy-langu.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'MTART'
      dynpprog     = sy-repid
      dynpnr       = sy-dynnr
      dynprofield  = 'GS_MAT-MTART'
      window_title = 'Mtart List'
      value_org    = 'S'
      display      = space
    TABLES
      value_tab    = lt_mtart.

ENDFORM.
