FORM posting .

  DATA    : lv_msg(100).

  CLEAR   : gs_bdcdata, gs_messtab, gs_param, lv_msg.
  REFRESH : gt_bdcdata, gt_messtab.

* Option 설정
  gs_param-dismode = 'N'.
  gs_param-nobinpt = 'X'.
  gs_param-updmode = 'S'.

  PERFORM make_bdcdata USING :
  'X'  'SAPBC402_PGCD_CREATE_CUSTOMER'  '0100',
  ' '  'BDC_OKCODE'                     '=SAVE',
  ' '  'SCUSTOM-NAME'                   gs_custom-name,
  ' '  'SCUSTOM-FORM'                   gs_custom-form,
  ' '  'SCUSTOM-STREET'                 gs_custom-street,
  ' '  'SCUSTOM-POSTBOX'                gs_custom-postbox,
  ' '  'SCUSTOM-POSTCODE'               gs_custom-postcode,
  ' '  'SCUSTOM-CITY'                   gs_custom-city,
  ' '  'SCUSTOM-COUNTRY'                gs_custom-country,
  ' '  'SCUSTOM-REGION'                 gs_custom-region,
  ' '  'SCUSTOM-TELEPHONE'              gs_custom-telephone,
  ' '  'SCUSTOM-EMAIL'                  gs_custom-email,
  ' '  'SCUSTOM-CUSTTYPE'               gs_custom-custtype,
  ' '  'SCUSTOM-DISCOUNT'               gs_custom-discount,
  ' '  'SCUSTOM-LANGU'                  gs_custom-langu.

  CALL TRANSACTION 'BC402_CALD_CRE_CUST'
             USING      gt_bdcdata " 인터널 테이블 주기
           OPTIONS FROM gs_param   " option 주기
          MESSAGES INTO gt_messtab." 메세지 받기

  READ TABLE gt_messtab INTO gs_messtab WITH KEY msgid = 'BC402_C62'
                                                 msgnr = '015'.

  IF sy-subrc = 0. " 성공했다면 아래 로직 수행.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = gs_messtab-msgid  " Message ID
        msgnr               = gs_messtab-msgnr  " Number of message
        msgv1               = gs_messtab-msgv1  " Parameter 1
        msgv2               = gs_messtab-msgv2  " Parameter 2
        msgv3               = gs_messtab-msgv3  " Parameter 3
        msgv4               = gs_messtab-msgv4  " Parameter 4
      IMPORTING
        message_text_output = lv_msg.           " Output message text

    CONDENSE lv_msg.

    MESSAGE s000 WITH lv_msg.

  ELSE.

    MESSAGE s000 WITH 'Posting failed' DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM make_bdcdata  USING pv_begin pv_fnam pv_fval.

  CASE pv_begin.
    WHEN 'X'.
      gs_bdcdata-dynbegin = pv_begin.
      gs_bdcdata-program  = pv_fnam.
      gs_bdcdata-dynpro   = pv_fval.

    WHEN OTHERS.
      gs_bdcdata-fnam = pv_fnam.
      gs_bdcdata-fval = pv_fval.

  ENDCASE.

  APPEND gs_bdcdata TO gt_bdcdata.
  CLEAR  gs_bdcdata.

ENDFORM.
