FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat = VALUE #(
                      key       = pv_key
                      fieldname = pv_field
                      coltext   = pv_text
                      ref_table = pv_ref_table
                      ref_field = pv_ref_field
                   ).

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM get_idol .

  _clear gs_idol gt_idol.

  SELECT groupid    AS group_id,    memberid     AS member_id,
         membername AS member_name, membergender AS member_gender,
         memberaddr AS member_addr, memberphone  AS member_phone,
         erdat,        erzet,      ernam,
         aedat,        aezet,      aenam
    INTO CORRESPONDING FIELDS OF TABLE @gt_idol
    FROM ZC519cds0015.

FORM display_data .

  DATA : lv_cnt TYPE i.

  CLEAR  lv_cnt.
  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH 'First, select row' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  lv_cnt = lines( gt_row ).

  IF lv_cnt > 1.

    MESSAGE s000 WITH 'Select only one row' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_idol INTO gs_idol INDEX gs_row-index.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc519t0030.

  IF gs_idol-group_id  IS INITIAL OR
     gs_idol-member_id IS INITIAL.

    MESSAGE s000 WITH 'Group ID, Member ID is required'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_idol TO ls_save.

  IF gs_idol-erdat IS INITIAL.

    gs_idol-erdat = sy-datum.
    gs_idol-erzet = sy-uzeit.
    gs_idol-ernam = sy-uname.

  ELSE.

    gs_idol-aedat = sy-datum.
    gs_idol-aezet = sy-uzeit.
    gs_idol-aenam = sy-uname.

  ENDIF.

  MODIFY ZC519t0030 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_idol.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM delete_data .

  DATA : lcl_statement TYPE REF TO cl_sql_statement,
         lv_sql        TYPE string,
         lx_sql_except TYPE REF TO cx_root,
         lv_msg        TYPE string.

  IF gs_idol-group_id  IS INITIAL OR
     gs_idol-member_id IS INITIAL.

    MESSAGE s000 WITH 'Group ID, Member ID is required'.
    EXIT.

  ENDIF.

  TRY.
* DB Connection 연결정보 얻기
      DATA(lcl_connection) = cl_sql_connection=>get_connection( ). "Default Connection 얻기

* SQL 을 실행할 객체에 Connection 전달
      CREATE OBJECT lcl_statement
        EXPORTING
          con_ref = lcl_connection. " Database Connection

* 실행할 SQL 구문 작성
      lv_sql = | DELETE FROM ZC519T0030                    | &&
               | WHERE GROUP_ID  = '{ gs_idol-group_id }'  | &&
               |   AND MEMBER_ID = '{ gs_idol-member_id }' | .

      lcl_statement->execute_update( lv_sql ).

      lcl_connection->commit( ).
      lcl_connection->close( ).

      PERFORM get_idol.
      PERFORM refresh_grid.

      MESSAGE s003.

    CATCH cx_sql_exception INTO lx_sql_except.

* 에러 메시지 받기
      lv_msg = lx_sql_except->get_text( ).
      lcl_connection->rollback( ).
      lcl_connection->close( ).

      " 위에서 에러 뜨면
      " 바로 catch문으로 오기 때문에 close 한 번 더 함.

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      EXIT.

  ENDTRY.

ENDFORM.
