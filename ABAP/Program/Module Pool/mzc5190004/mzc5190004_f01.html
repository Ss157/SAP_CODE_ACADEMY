  PERFORM make_fcat USING :
  'X'  'AUFNR'  ' '  'ZC519T0041'  ' ',

  ' '  'PWERK'  ' '  'ZC519T0041'  ' ',
  ' '  'GSTRP'  ' '  'ZC519T0041'  ' ',
  ' '  'GLTRP'  ' '  'ZC519T0041'  ' ',

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

* AUFNR 필드에 핫스팟을 주기 위해서는
* 필드 카탈로그에서 아래와 같이 X 표시를 주어야 한다.
  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'GAMNG'.
      gs_fcat-qfieldname = 'GMEIN'.

    WHEN 'AUFNR'.
      gs_fcat-hotspot    = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_data .

* lines를 사용해서 선택한 행의 개수를 보고 다중 행이면
* 메세지 처리와 exit를 해주었다.
* 그 후, gt_row에서 선택한 행의 index를 읽었다.
* 얻은 index를 사용해서 gt_order에서 gs_order로 값을 읽어
* display했다.
  DATA : lv_cnt TYPE i.

  CLEAR  lv_cnt.
  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  lv_cnt = lines( gt_row ).

  IF lv_cnt > 1.

    MESSAGE s000 WITH TEXT-f02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_order INTO gs_order INDEX gs_row-index.

ENDFORM.

FORM save_data .

* 하나의 행을 save하는 것이기 때문에 lt_save는 선언할
* 필요가 없어서 ls_save만 선언한다.
* pk 중 하나라도 빈 값이면 메세지 처리하고 exit한다.
* pk가 정상적으로 들어왔다면 erdat가 빈 값인지 확인하고
* 그에 따라 timestamp를 관리한다.
* modify 한 후 성공적이였다면 commit 한 후
* select와 grid를 다시 그림으로써 화면상에서 사용자가
* 인지할 수 있게 해준다.
* modify 성공 여부에 따라서 메세지 처리를 해준다.
  DATA : ls_save TYPE zc519t0041.

  CLEAR  ls_save.

  IF gs_order-aufnr IS INITIAL OR
     gs_order-posnr IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_order TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc519t0041 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_order.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM delete_data .

* pk 중 하나라도 빈 값이면 메세지 처리하고 exit한다.
* pk 값에 따라 delete 한 후 성공적이였다면 commit 한 후
* select와 grid를 다시 그림으로써 화면상에서 사용자가
* 인지할 수 있게 해준다.
* delete 성공 여부에 따라서 메세지 처리를 해준다.
  IF gs_order-aufnr IS INITIAL OR
     gs_order-posnr IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc519t0041
   WHERE aufnr = gs_order-aufnr
     AND posnr = gs_order-posnr.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_order.
    PERFORM refresh_grid.

    MESSAGE s003.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM get_order .

  _clear gs_order gt_order.

  SELECT aufnr posnr matnr pwerk gstrp
         gltrp rsnum gamng gmein
         erdat erzet ernam
         aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_order
    FROM zc519t0041
   ORDER BY aufnr posnr.

ENDFORM.

FORM handle_hotspot  USING ps_col TYPE lvc_s_col
                           ps_row TYPE lvc_s_row.

* rowtype이 0이 아니면 exit한다.
* 일반 행은 rowtype이 0이고 합계, 소계 등은
* rowtype이 0이 아니기 때문이다.
* aufnr을 클릭했으면 파리미터 아이디를 설정하고
* 트랜잭션을 부른다.
* 또한, 첫 화면은 건너뛰어준다.
  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_order INTO gs_order INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'AUFNR'.

      SET PARAMETER ID 'ANR' FIELD gs_order-aufnr.

      CALL TRANSACTION 'CO03' AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.

FORM handle_double_click  USING ps_col TYPE lvc_s_col
                                ps_row TYPE lvc_s_row.

* rowtype이 0이 아니면 exit한다.
* 일반 행은 rowtype이 0이고 합계, 소계 등은
* rowtype이 0이 아니기 때문이다.
* matnr을 클릭했으면 파리미터 아이디를 설정하고
* 트랜잭션을 부른다.
* 또한, 첫 화면은 건너뛰어준다.
  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_order INTO gs_order INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.

      SET PARAMETER ID 'MAT' FIELD gs_order-matnr.

      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.
