FORM display_text_edit .

  CREATE OBJECT gcl_container_l
    EXPORTING
      container_name = 'GCL_CONTAINER_L'.

  CREATE OBJECT gcl_container_r
    EXPORTING
      container_name = 'GCL_CONTAINER_R'.

  CREATE OBJECT gcl_text_l
    EXPORTING
      parent        = gcl_container_l
      wordwrap_mode = gcl_text_l->wordwrap_at_windowborder.
  " 윈도우 창 끝에 만나면 자동으로 줄바꿈되어 화면에 표시되는 것

  CREATE OBJECT gcl_text_r
    EXPORTING
      parent        = gcl_container_r
      wordwrap_mode = gcl_text_r->wordwrap_at_windowborder.

* Toolbar 제거
  CALL METHOD gcl_text_l->set_toolbar_mode
    EXPORTING
      toolbar_mode = 0.

* Statusbar 제거
  CALL METHOD gcl_text_l->set_statusbar_mode
    EXPORTING
      statusbar_mode = 0.

* 읽기전용
  CALL METHOD gcl_text_r->set_readonly_mode
    EXPORTING
      readonly_mode = 1.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM save_text_table .

  DATA : ls_save TYPE zc519t0042,

         BEGIN OF ls_text,
           line TYPE tdline,
         END OF ls_text,

         lt_text LIKE TABLE OF ls_text.

  CLEAR   : ls_text, ls_save.
  REFRESH : lt_text.

  IF gv_seq IS INITIAL.

    MESSAGE s000 WITH 'Input Sequence' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  CALL METHOD gcl_text_l->get_text_as_r3table
    IMPORTING
      table               = lt_text
    EXCEPTIONS
      potential_data_loss = 1.

  IF lt_text IS INITIAL.

    MESSAGE s000 WITH 'Input Description' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  ls_save-seq = gv_seq.

  LOOP AT lt_text INTO ls_text.

    CASE sy-tabix.
      WHEN 1.
        CONCATENATE ls_save-description ls_text-line INTO ls_save-description.

      WHEN OTHERS.
        CONCATENATE ls_save-description ls_text-line INTO ls_save-description
        SEPARATED BY cl_abap_char_utilities=>newline.

    ENDCASE.

  ENDLOOP.

  MODIFY zc519t0042 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.
    MESSAGE s002.

  ENDIF.

* 문자열 자르기
* ls_save-description+0(3) : 맨앞에서 3개의 문자만 자르겠다.

ENDFORM.

*& Form read_text_table

*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM read_text_table .

  DATA : ls_data TYPE zc519t0042,

         BEGIN OF ls_text,
           line TYPE tdline,
         END OF ls_text,

         lt_text LIKE TABLE OF ls_text.

  CLEAR   : ls_text, ls_data.
  REFRESH : lt_text.

  IF gv_seq IS INITIAL.

    MESSAGE s000 WITH 'Input Sequence' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SELECT SINGLE seq description
    INTO CORRESPONDING FIELDS OF ls_data
    FROM zc519t0042
   WHERE seq = gv_seq.

  IF sy-subrc <> 0.

    MESSAGE s001.
    EXIT.

  ENDIF.

  IF ls_data-description IS INITIAL.
    EXIT.
  ENDIF.

  "문자열 나누기 : SPLIT.. AT.. INTO
  " 1. SPLIT <변수> AT <구분자> INTO <변수1> <변수2> ... <변수n>.
  " -> 구분자에 의해서 나눠지는 문자열들을 차례대로 <변수1> ~ <변수n> 까지 넣어줌.
  "    INTO 이후에 나열된 변수보다 많은 경우 나머지 문자열은 무시됨
  "    나눠지는 문자열이 나열된 변수보다 적은 경우 나머지 변수 무시됨
  " 2. SPLIT <변수> AT <구분자> INTO TABLE <ITAB>
  " -> 구분자에 의해서 나눠지는 문자열들을 <ITAB>에 차례대로 APPEND 해줌
  "    몇 개로 받을지 예측 불가능할 때 인터널 테이블로 SPLIT함

  "text edit로 올라갈 때 인터널 테이블로 올라간다.
  SPLIT ls_data-description AT cl_abap_char_utilities=>newline INTO TABLE lt_text.

  CALL METHOD gcl_text_r->set_text_as_r3table
    EXPORTING
      table = lt_text.

ENDFORM.
