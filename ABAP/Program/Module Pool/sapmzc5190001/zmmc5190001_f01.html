FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      container_name = 'GCL_CONTAINER'.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_input
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM display_data .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_input INTO gs_input INDEX gs_row-index.

ENDFORM.

FORM get_student .

  _clear gs_input gt_input.

  SELECT class cl_num name  phone gender
         erdat erzet  ernam aedat aezet
         aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_input
    FROM zc519t0002
   ORDER BY class cl_num.

FORM save_data .

  DATA : ls_save TYPE zc519t0002.

  CLEAR  ls_save.

  "PK 필수값 체크
  IF gs_input-class  IS INITIAL OR
     gs_input-cl_num IS INITIAL.

    MESSAGE s000 WITH TEXT-f02 "Class and Student No is required field
    DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_input TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc519t0002 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_student.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM delete_data .

  IF gs_input-class  IS INITIAL OR
     gs_input-cl_num IS INITIAL.

    MESSAGE s000 WITH TEXT-f02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc519t0002
   WHERE class  = gs_input-class
     AND cl_num = gs_input-cl_num.

  IF sy-subrc = 0.

    COMMIT WORK AND WAIT.

    PERFORM get_student.
    PERFORM refresh_grid.

    MESSAGE s003.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.
