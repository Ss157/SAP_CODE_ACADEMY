FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PRODUCT_QTY'.
      gs_fcat-qfieldname = 'PRODUCT_UNIT'.

    WHEN 'UNIT_PRICE'.
      gs_fcat-cfieldname = 'PRICE_UNIT'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      container_name = 'GCL_CONTAINER'.

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container
      rows    = 2
      columns = 1.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_text.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 2
      column    = 1
    RECEIVING
      container = gcl_container_alv.

*  Text Edit 높이 설정
  CALL METHOD gcl_splitter->set_row_height
    EXPORTING
      id     = 1
      height = 30.

* 위 컨테이너에 Text Edit 설정
  CREATE OBJECT gcl_text
    EXPORTING
      wordwrap_mode = gcl_text->wordwrap_at_windowborder
      parent        = gcl_container_text.

*  Text Edit Toolbar Statusbar 없애기
  CALL METHOD gcl_text->set_toolbar_mode
    EXPORTING
      toolbar_mode = 0.

  CALL METHOD gcl_text->set_statusbar_mode
    EXPORTING
      statusbar_mode = 0.

* 아래 컨테이너에 ALV 설정
  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container_alv.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_plant
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM f4_werks .

  DATA : BEGIN OF lt_werks OCCURS 0,
           werks TYPE t001w-werks,
           name1 TYPE t001w-name1,
         END OF lt_werks.

*         lt_werks LIKE TABLE OF ls_werks.

  _clear lt_werks lt_werks.

  SELECT werks name1
    INTO CORRESPONDING FIELDS OF TABLE lt_werks
    FROM t001w.

* werks에 대한 서치 헬프 생성
  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'WERKS'       "입력 값        " Key for personal help
      dynpprog     = sy-repid                      " Current program
      dynpnr       = sy-dynnr                      " Screen number
      dynprofield  = 'GS_PLANT-WERKS' " 값 넣을 곳 " Name of screen field for value return
      window_title = 'Plant List'                  " Title for the hit list
      value_org    = 'S'
      display      = space
    TABLES
      value_tab    = lt_werks[].

ENDFORM.

FORM display_data .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

* ALV의 행이 선택되지 않았다면 메세지 발생
  IF gt_row IS INITIAL.

    MESSAGE s000 WITH 'First, select a row' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_plant INTO gs_plant INDEX gs_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

* DB의 Memo 필드를 Text Edit에 나오게 설정
  CALL METHOD gcl_text->set_textstream
    EXPORTING
      text = gs_plant-memo.

ENDFORM.

FORM save_data .

  DATA :  ls_save TYPE zc519t0043.

  CLEAR : ls_save.

* 필수 값 확인
  IF gs_plant-steel IS INITIAL OR
     gs_plant-werks IS INITIAL.

    MESSAGE s000 WITH 'Steel, Plant are required fields'
    DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_plant TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

* Text Edit에 입력된 값을 DB에 저장할 스트럭쳐에 넣기
  CALL METHOD gcl_text->get_textstream
    IMPORTING
      text = ls_save-memo.

  cl_gui_cfw=>flush( ).

  MODIFY zc519t0043 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_plant.
    PERFORM refresh_grid.

    CALL METHOD gcl_text->delete_text.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM get_plant .

  _clear gs_plant gt_plant.

  SELECT steel       werks        industry_kind area       memo
         product_qty product_unit unit_price    price_unit
         erdat       erzet        ernam
         aedat       aezet        aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_plant
    FROM zc519t0043
   ORDER BY steel werks.

ENDFORM.

FORM delete_data .

  IF gs_plant-steel IS INITIAL OR
     gs_plant-werks IS INITIAL.

    MESSAGE s000 WITH 'First, select row' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc519t0043
   WHERE steel = gs_plant-steel
     AND werks = gs_plant-werks.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_plant.
    PERFORM refresh_grid.

    CALL METHOD gcl_text->delete_text.

    MESSAGE s003.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.
