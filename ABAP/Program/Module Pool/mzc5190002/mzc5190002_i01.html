MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'SAVE'.
      CLEAR gv_okcode.

      PERFORM save_data.

    WHEN 'DELETE'.
      CLEAR gv_okcode.

      PERFORM delete_data.

    WHEN 'DISPLAY'.
      CLEAR gv_okcode.

      PERFORM display_data.

    WHEN 'CLEAR'.
      CLEAR gv_okcode.

      PERFORM clear_data.

    WHEN 'SETDESC'.
      CLEAR gv_okcode.

      PERFORM set_desc.

  ENDCASE.

ENDMODULE.
