FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'LABST'.
      gs_fcat-qfieldname = 'MEINS'.

    WHEN 'NETPR'.
      gs_fcat-cfieldname = 'WAERS'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_data .
* ZC519R0051_F01 <-- lines
  DATA : lv_cnt TYPE i.

  CLEAR  lv_cnt.
  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL .

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  lv_cnt = lines( gt_row ).

  IF lv_cnt <> 1.

    MESSAGE s000 WITH TEXT-f02 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  READ TABLE gt_goods INTO gs_goods INDEX gs_row-index.

ENDFORM.

FORM get_goods .

  _clear gs_goods gt_goods.

  SELECT goods category goods_count goods_desc
         labst meins    netpr       waers
         erdat erzet    ernam
         aedat aezet    aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_goods
    FROM zc519t0039
   ORDER BY goods.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc519t0039.

  CLEAR  ls_save.

  IF gs_goods-goods IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  MOVE-CORRESPONDING gs_goods TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc519t0039 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.

    PERFORM get_goods.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM f4_goods .

  DATA : ls_goods  TYPE zc519t0014,
         lt_goods  LIKE TABLE OF ls_goods,

         ls_return TYPE ddshretval,
         lt_return LIKE TABLE OF ls_return.

  _clear ls_goods  lt_goods.
  _clear ls_return lt_return.

  SELECT goods goods_desc
    INTO CORRESPONDING FIELDS OF TABLE lt_goods
    FROM zc519t0014.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'GOODS'                " Name of return field in FIELD_TAB
      dynpprog     = sy-repid            " Current program
      dynpnr       = sy-dynnr            " Screen number
      dynprofield  = 'GS_GOODS-GOODS'            " Name of screen field for value return
      window_title = 'Goods Description'                " Title for the hit list
      value_org    = 'S'              " Value return: C: cell by cell, S: structured
      display      = space            " Override readiness for input
    TABLES
      value_tab    = lt_goods                 " Table of values: entries cell by cell
*     field_tab    =                  " Fields of the hit list
      return_tab   = lt_return.

  READ TABLE lt_return INTO ls_return INDEX 1.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  gs_goods-goods = ls_return-fieldval.

  READ TABLE lt_goods INTO ls_goods WITH KEY goods = gs_goods-goods.

  IF sy-subrc = 0.

    CALL FUNCTION 'SAPGUI_SET_FUNCTIONCODE'
      EXPORTING
        functioncode           = 'SETDESC'
      EXCEPTIONS
        function_not_supported = 1
        OTHERS                 = 2.

  ENDIF.

ENDFORM.

FORM set_desc.

  CLEAR gs_goods-goods_desc.

  SELECT SINGLE goods_desc
    INTO gs_goods-goods_desc
    FROM zc519t0014
   WHERE goods = gs_goods-goods.

ENDFORM.
