FORM get_flight .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~carrid a~carrname
         b~connid b~airpfrom b~airpto   b~distance b~distid
         c~fldate c~price    c~currency
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM scarr      AS a
   INNER JOIN spfli AS b
      ON a~carrid = b~carrid
   INNER JOIN sflight AS c
      ON b~carrid = c~carrid
     AND b~connid = c~connid
   WHERE a~carrid IN so_carr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PRICE'.
      gs_fcat-cfieldname = 'CURRENCY'. "인터널테이블의 통화키 필드이름

    WHEN 'DISTANCE'.
      gs_fcat-qfieldname = 'DISTID'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid. "sy-cprog 도 같음. 리포트 id를 넣어줌.
  "메인프로그램 ID를 넣어줌.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
