FORM init_parm .

  so_bukrs-sign   = 'I'.
  so_bukrs-option = 'EQ'.
  so_bukrs-low    = '0001'.

  APPEND so_bukrs.

  so_gjahr-sign   = 'I'.
  so_gjahr-option = 'EQ'.
  so_gjahr-low    = '2010'.

  APPEND so_gjahr.

ENDFORM.

FORM get_account .

  CLEAR   : gs_account.
  REFRESH : gt_account.

  SELECT a~bukrs a~gjahr a~belnr a~blart a~budat
         b~buzei b~bschl b~koart b~shkzg b~dmbtr b~pswsl
         c~ltext AS blart_txt
         d~ltext AS bschl_txt
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM      bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart = c~blart
     AND c~spras = sy-langu
    LEFT OUTER JOIN tbslt AS d
      ON b~bschl = d~bschl
     AND d~spras = sy-langu
   WHERE a~bukrs IN so_bukrs
     AND a~gjahr IN so_gjahr.

  IF sy-subrc <> 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM modify_account .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_account INTO gs_account.

    lv_tabix = sy-tabix.

    CASE gs_account-shkzg.
      WHEN 'H'.
        gs_account-dmbtr *= -1.

        MODIFY gt_account FROM gs_account INDEX lv_tabix
        TRANSPORTING dmbtr.

    ENDCASE.

  ENDLOOP.

ENDFORM.
