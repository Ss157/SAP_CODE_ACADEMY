FORM get_airline .

  DATA : lv_code,
         lv_msg(100),
         lcl_airline TYPE REF TO zc519cl001.

  CLEAR   : lv_code, lv_msg.
  REFRESH : gt_data.

*  CALL FUNCTION 'ZC519FM_GET_AIRLINE_INFO'
*    EXPORTING
*      iv_carrid = pa_carr
*    IMPORTING
*      ev_code   = lv_code
*      ev_msg    = lv_msg
*    TABLES
*      et_data   = gt_data.

  IF lcl_airline IS NOT BOUND.

    CREATE OBJECT lcl_airline.

  ENDIF.

  CALL METHOD lcl_airline->get_airline_info
    EXPORTING
      iv_carrid = pa_carr
    IMPORTING
      ev_code   = lv_code
      ev_msg    = lv_msg
    CHANGING
      et_data   = gt_data.

* gt_data IS INITIAL인지로 판단할 수도 있다.
  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    STOP.

  ENDIF.

ENDFORM.

*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PRICE' OR 'PAYMENTSUM'.
      gs_fcat-cfieldname = 'CURRENCY'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
