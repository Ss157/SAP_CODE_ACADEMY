*& -->  p1        text
*& <--  p2        text

FORM get_account .

  _clear gs_account gt_account.

  SELECT a~blart a~budat
         b~belnr b~buzei b~bschl b~koart b~gsber
         b~shkzg b~hkont b~dmbtr b~pswsl
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM bkpf      AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
   WHERE b~bukrs = pa_bukrs
     AND b~gjahr = pa_gjahr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.

    WHEN 'BELNR'.
      gs_fcat-hotspot    = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid,
                lcl_event_handler=>handle_hotspot      FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_account
      it_fieldcatalog = gt_fcat.

ENDFORM.

*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_account INTO gs_account INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'BELNR'.
      SET PARAMETER ID : 'BLN' FIELD gs_account-belnr,
                         'BUK' FIELD pa_bukrs,
                         'GJR' FIELD pa_gjahr.

      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.

*& Form handle_double_click
*&---------------------------------------------------------------------*

*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_account INTO gs_account INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'HKONT'.
      SET PARAMETER ID : 'ACC' FIELD gs_account-hkont,
                         'BUK' FIELD pa_bukrs,
                         'GJR' FIELD pa_gjahr.

      CALL TRANSACTION 'FAGLB03' AND SKIP FIRST SCREEN.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.
