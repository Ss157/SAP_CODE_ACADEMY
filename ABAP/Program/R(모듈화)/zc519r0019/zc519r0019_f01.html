FORM get_account .

  CLEAR   : gs_account.
  REFRESH : gt_account.

  SELECT a~blart a~budat
         b~belnr b~buzei b~bschl b~koart b~gsber b~shkzg b~dmbtr b~pswsl
         c~ltext
         d~gtext
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM bkpf      AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart = c~blart
     AND c~spras = sy-langu
    LEFT OUTER JOIN tgsbt AS d
      ON b~gsber = d~gsber
     AND d~spras = sy-langu
   WHERE a~bukrs  = pa_bukrs
     AND a~gjahr  = pa_gjahr
     AND a~belnr IN so_belnr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

      gs_fcat-icon = 'X'.

    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.
      gs_fcat-do_sum     = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

FORM display_alv .

    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_account
      it_fieldcatalog = gt_fcat

FORM set_icon .

  DATA : lv_tabix TYPE sy-tabix,
         ls_color TYPE lvc_s_scol.
  CLEAR  lv_tabix.

  LOOP AT gt_account INTO gs_account.

    lv_tabix = sy-tabix.

    CLEAR   : ls_color.
    REFRESH : gs_account-color.

    CASE gs_account-shkzg.
      WHEN 'S'.
        gs_account-status = icon_positive.

        MODIFY gt_account FROM gs_account INDEX lv_tabix
        TRANSPORTING status.

      WHEN 'H'.
        gs_account-status = icon_negative.

        ls_color-fname     = 'DMBTR'.
        ls_color-color-col = 6.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

        APPEND ls_color TO gs_account-color.
        CLEAR  ls_color.

        MODIFY gt_account FROM gs_account INDEX lv_tabix
        TRANSPORTING status color.

    ENDCASE.

  ENDLOOP.

ENDFORM.
