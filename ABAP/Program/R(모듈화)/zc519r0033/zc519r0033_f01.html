FORM get_flight .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT carrid connid     fldate   planetype
         price  paymentsum currency
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM sflight
   WHERE carrid IN so_carr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra      = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-row = 'X'.
  gs_stable-col = 'X'.

  PERFORM make_fcat USING :
          'X'  'CARRID'     ' '  'SFLIGHT'  ' ',
          'X'  'CONNID'     ' '  'SFLIGHT'  ' ',
          'X'  'FLDATE'     ' '  'SFLIGHT'  ' ',
          ' '  'PLANETYPE'  ' '  'SFLIGHT'  ' ',
          ' '  'PRICE'      ' '  'SFLIGHT'  ' ',
          ' '  'PAYMENTSUM' ' '  'SFLIGHT'  ' ',
          ' '  'CURRENCY'   ' '  'SFLIGHT'  ' '.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'CARRID' OR 'PLANETYPE'.
      gs_fcat-hotspot = 'X'.

    WHEN 'PRICE'.
      gs_fcat-cfieldname = 'CURRENCY'.

    WHEN 'PAYMENTSUM'.
      gs_fcat-cfieldname = 'CURRENCY'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER lcl_event_handler=>handle_hotspot FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.

    EXIT.

  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'CARRID'.
      PERFORM get_carrid USING ps_row.

    WHEN 'PLANETYPE'.
      PERFORM modify_planetype USING ps_row.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.

FORM get_carrid  USING ps_row TYPE lvc_s_row.

  ENDIF.

ENDFORM.
