FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT company   plant    storage material    mat_desc
         serial_no quantity unit    unit_price  price_key
         use_yn
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM zc519t0034
   WHERE company EQ pa_comp
     AND plant   IN so_plant
     AND storage IN so_stora.

  IF sy-subrc NE 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_sort         = gt_sort
      it_fieldcatalog = gt_fcat.

ENDFORM.

*&---------------------------------------------------------------------*

*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'USE_YN'.
      gs_fcat-checkbox = 'X'.

    WHEN 'QUANTITY'.
      gs_fcat-qfieldname = 'UNIT'.
      gs_fcat-do_sum     = 'X'.

    WHEN 'UNIT_PRICE'.
      gs_fcat-cfieldname = 'PRICE_KEY'.
      gs_fcat-do_sum     = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
