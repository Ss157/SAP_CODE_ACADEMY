** 개발을 진행하다 보니.. ALV 출력용 itab gt_data 기준으로 sflight 를
* select 할 상황 발생..
* for all entries 로 붙어야 하는데 이것을 사용하기 위해서는 중복제거가 필수..
* 중복제거 하다가는 gt_data 의 레코드가 손실이 될 위험이 생기므로..
* for all entries 로 JOIN 할 필드의 레코드만 따로 빼서 써야 함
* 여기서는 gt_data 의 carrid, connid 기준으로 sflight 를 select 할 상황이라 하자고..

* for all entries 로 JOIN 에 사용할 itab을 선언
DATA : BEGIN OF ls_key,
         carrid TYPE sflight-carrid,
         connid TYPE sflight-connid,
       END OF ls_key,
       lt_key     LIKE TABLE OF ls_key,

       lt_sflight TYPE TABLE OF sflight.

* carrid, connid 필드 레코드만 lt_key 로 보냄
MOVE-CORRESPONDING gt_data TO lt_key.

SORT lt_key BY carrid connid.
DELETE ADJACENT DUPLICATES FROM lt_key COMPARING carrid connid.

IF lt_key IS NOT INITIAL.

  SELECT carrid connid fldate price currency
    INTO CORRESPONDING FIELDS OF TABLE lt_sflight
    FROM sflight
     FOR ALL ENTRIES IN lt_key
   WHERE carrid   = lt_key-carrid
     AND connid   = lt_key-connid
     AND currency = 'USD'.

ENDIF.

******************ITAB 을 이용한 Group by 및 SUM, COUNT ***********************
* GROUP BY 의 효과 : 중복제거의 효과가 있음

DATA : ls_flight TYPE sflight,
       lt_flight LIKE TABLE OF ls_flight,

       BEGIN OF ls_sum,
         carrid TYPE sflight-carrid,
         connid TYPE sflight-connid,
         price  TYPE sflight-price,
       END OF ls_sum,

       lt_sum LIKE TABLE OF ls_sum,

       BEGIN OF ls_total_sum,
         price TYPE sflight-price,
       END OF ls_total_sum,

       lt_total_sum LIKE TABLE OF ls_total_sum,

       BEGIN OF ls_count,
         carrid TYPE sflight-carrid,
         cnt    TYPE i,
       END OF ls_count,

       lt_count LIKE TABLE OF ls_count,

       BEGIN OF ls_total_count,
         cnt TYPE i,
       END OF ls_total_count,

       lt_total_count LIKE TABLE OF ls_total_count,

       BEGIN OF ls_dup,
         carrid TYPE sflight-carrid,
         connid TYPE sflight-connid,
       END OF ls_dup,

       lt_dup LIKE TABLE OF ls_dup.

SELECT *
  INTO CORRESPONDING FIELDS OF TABLE lt_flight
  FROM sflight
 ORDER BY carrid connid fldate.

* CARRID, CONNID 별로 Price의 SUM 을 ITAB에 구성하기
LOOP AT lt_flight INTO ls_flight.

* price 를 전체 SUM 해보자고
* 효과
*    SELECT SUM( price )
*      FROM sflight.
  ls_total_sum-price = ls_flight-price.

  COLLECT ls_total_sum INTO lt_total_sum.

* CARRID, CONNID 별 price 를 SUM 해보자고..
* 효과
*    SELECT carrid connid SUM( price )
*      FROM sflight
*     GROUP BY carrid connid

  ls_sum-carrid = ls_flight-carrid.
  ls_sum-connid = ls_flight-connid.
  ls_sum-price  = ls_flight-price.

  COLLECT ls_sum INTO lt_sum.

* 레코드 전체를 COUNT 해보자고
* 효과
*    SELECT COUNT( * )
*      FROM sflight.

  ls_total_count-cnt = 1.

  COLLECT ls_total_count INTO lt_total_count.

* CARRID 별 COUNT를 해보자고..
* 효과
*    SELECT carrid COUNT( * )
*      FROM sflight
*     GROUP BY carrid.

  ls_count-carrid = ls_flight-carrid.
  ls_count-cnt    = 1.

  COLLECT ls_count INTO lt_count.

* CARRID, CONNID 기준으로 중복제거 해보자고..
* 효과
*    SELECT carrid connid
*      FROM sflight
*     GROUP BY carrid connid.

  ls_dup-carrid = ls_flight-carrid.
  ls_dup-connid = ls_flight-connid.

  COLLECT ls_dup INTO lt_dup.

ENDLOOP.
