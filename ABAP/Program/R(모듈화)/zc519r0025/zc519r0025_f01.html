FORM get_text .

  DATA : lv_tabix    TYPE sy-tabix,
         lv_code,
         lv_msg(100).
  CLEAR : lv_tabix, lv_code, lv_msg.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    CLEAR : lv_code, lv_msg.

    call function 'ZC519FM_GET_BLART_TEXT'
      EXPORTING
        pi_blart = gs_data-blart
        pi_spras = sy-langu
      IMPORTING
        pe_ltext = gs_data-ltext
        pe_code  = lv_code
        pe_msg   = lv_msg.

    CASE lv_code.
      WHEN 'S'.
        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING ltext.

    ENDCASE.

    CLEAR : lv_code, lv_msg.

    call function 'ZC519FM_GET_GSBER_TEXT'
      EXPORTING
        pi_gsber = gs_data-gsber
        pi_spras = sy-langu
      IMPORTING
        pe_gtext = gs_data-gtext
        pe_code  = lv_code
        pe_msg   = lv_msg.

    CASE lv_code.
      WHEN 'S'.
        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING gtext.

    ENDCASE.

  ENDLOOP.

ENDFORM.

*&      --> P_
*&      --> P_

*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
