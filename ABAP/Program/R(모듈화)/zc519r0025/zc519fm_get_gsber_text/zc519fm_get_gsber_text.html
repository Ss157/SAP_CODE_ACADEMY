FUNCTION zc519fm_get_gsber_text.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_GSBER) TYPE  GSBER
*"     REFERENCE(PI_SPRAS) TYPE  SPRAS OPTIONAL
*"  EXPORTING
*"     REFERENCE(PE_GTEXT) TYPE  GTEXT
*"     REFERENCE(PE_CODE) TYPE  CHAR1
*"     REFERENCE(PE_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lv_spras TYPE sy-langu.

  IF pi_spras IS INITIAL.

    lv_spras = sy-langu.

  ELSE.

    lv_spras = pi_spras.

  ENDIF.

  SELECT SINGLE gtext
    INTO pe_gtext
    FROM tgsbt
   WHERE spras = lv_spras
     AND gsber = pi_gsber.

  IF sy-subrc <> 0.

    pe_code = 'E'.
    pe_msg  = 'Data Not Found'.

  ELSE.

    pe_code = 'S'.

  ENDIF.

ENDFUNCTION.
