FUNCTION zc519fm_get_blart_text.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_BLART) TYPE  BLART
*"     REFERENCE(PI_SPRAS) TYPE  SPRAS DEFAULT SY-LANGU
*"  EXPORTING
*"     REFERENCE(PE_LTEXT) TYPE  LTEXT_003T
*"     REFERENCE(PE_CODE) TYPE  CHAR1
*"     REFERENCE(PE_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE ltext
    INTO pe_ltext
    FROM t003t
   WHERE spras = pi_spras
     AND blart = pi_blart.

  IF sy-subrc <> 0.

    pe_code = 'E'.
    pe_msg  = 'Data Not Found'.

  ELSE.

    pe_code = 'S'.

  ENDIF.

ENDFUNCTION.
