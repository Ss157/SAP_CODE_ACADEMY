FORM get_airline .

  RANGES : lr_carrid FOR scarr-carrid,
           lr_connid FOR sflight-connid,
           lr_fldate FOR sflight-fldate.

  _clear lr_carrid  lr_carrid.
  _clear lr_connid  lr_connid.
  _clear lr_fldate  lr_fldate.
  _clear gs_airline gt_airline.

  IF pa_carr IS NOT INITIAL.

    lr_carrid-sign   = 'I'.
    lr_carrid-option = 'EQ'.
    lr_carrid-low    = pa_carr.

    APPEND lr_carrid.

  ENDIF.

  IF pa_conn IS NOT INITIAL.

    lr_connid-sign   = 'I'.
    lr_connid-option = 'EQ'.
    lr_connid-low    = pa_conn.

    APPEND lr_connid.

  ENDIF.

  IF pa_flda IS NOT INITIAL.

    lr_fldate-sign   = 'I'.
    lr_fldate-option = 'EQ'.
    lr_fldate-low    = pa_flda.

    APPEND lr_fldate.

  ENDIF.

  SELECT status,   carrid, connid, fldate,  carrname,
         cityfrom, cityto, price,  currcode
    INTO CORRESPONDING FIELDS OF TABLE @gt_airline
    FROM zc519cds0013
   WHERE carrid IN @lr_carrid
     AND connid IN @lr_connid
     AND fldate IN @lr_fldate .

  IF sy-subrc <> 0.

    MESSAGE s001 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.

*& Form make_fcat
*&---------------------------------------------------------------------*

*&      --> P_
*&      --> P_

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat = VALUE #(
                      key       = pv_key
                      fieldname = pv_field
                      coltext   = pv_text
                      ref_table = pv_ref_table
                      ref_field = pv_ref_field
                   ).

  CASE pv_field.
    WHEN 'PRICE'.
      gs_fcat = VALUE #( BASE gs_fcat
                         cfieldname = 'CURRCODE'
                       ).

    WHEN 'STATUS'.
      gs_fcat = VALUE #( BASE gs_fcat
                         icon = 'X'
                       ).

    WHEN 'CARRID'.
      gs_fcat = VALUE #( BASE gs_fcat
                         hotspot = 'X'
                       ).

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM handle_hotspot_click USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_airline INTO gs_airline INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'CARRID'.
      PERFORM get_sbook USING gs_airline-carrid
                              gs_airline-connid
                              gs_airline-fldate.

      IF gt_airline IS INITIAL.

        MESSAGE s000 WITH 'Flight Booking is not found'.
        EXIT.

      ENDIF.

      CALL SCREEN '0101' STARTING AT 20 3.

  ENDCASE.

ENDFORM.

*&      --> GS_AIRLINE_CARRID
*&      --> GS_AIRLINE_CONNID
*&      --> GS_AIRLINE_FLDATE
*&---------------------------------------------------------------------*
FORM get_sbook  USING pv_carrid TYPE sbook-carrid
                      pv_connid TYPE sbook-connid
                      pv_fldate TYPE sbook-fldate.

  DATA : lcl_statement TYPE REF TO cl_sql_statement,
*         lcl_connection TYPE REF TO cl_sql_connection,
*         lcl_result    TYPE REF TO cl_sql_result_set, " 결과 받을 곳
         lr_data       TYPE REF TO data, " any data 어떤 데이터든지 될 수 있음
         lv_sql        TYPE string,
         lx_sql_except TYPE REF TO cx_root,
         lv_msg        TYPE string.

  _clear gs_sbook gt_sbook.

  TRY.
* DB Connection 연결정보 얻기
      DATA(lcl_connection) = cl_sql_connection=>get_connection( ). "Default Connection 얻기

* SQL 을 실행할 객체에 Connection 전달
      CREATE OBJECT lcl_statement
        EXPORTING
          con_ref = lcl_connection. " Database Connection

* 실행할 SQL 구문 작성
      lv_sql = | SELECT CARRID,  CONNID,     FLDATE,    | &&
               |        BOOKID,  LUGGWEIGHT, WUNIT,     | &&
               |        INVOICE, FORCURAM,   FORCURKEY  | &&
               |   FROM SBOOK                           | &&
               |  WHERE MANDT  = '{ sy-mandt }'         | &&
               |    AND CARRID = '{ pv_carrid }'        | &&
               |    AND CONNID = '{ pv_connid }'        | &&
               |    AND FLDATE = '{ pv_fldate }'        |.

* SQL 을 실행하고 그 결과를 받아라..
      DATA(lcl_result) = lcl_statement->execute_query( lv_sql ).

* 결과를 받기 위해서 Any Data와 itab을 동기화 시킴( Call by reference 처럼 )
      GET REFERENCE OF gt_sbook INTO lr_data.

* 결과를 받으러 lr_data 가 DB로 들어감.
      lcl_result->set_param_table( lr_data ).

* 결과를 lr_data가 받았으니 이를 gt_sflight 와 공유
* -> 이순간에 gt_data 에 레코드 들어감
      lcl_result->next_package( ).

* Connection 자원 반환
      lcl_result->close( ).

    CATCH cx_sql_exception INTO lx_sql_except.

* 에러 메시지 받기
      lv_msg = lx_sql_except->get_text( ).
      lcl_result->close( ).
      " lcl_result->next_package( ).에서 에러 뜨면
      " 바로 catch문으로 오기 때문에 close 한 번 더 함.

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      EXIT.

  ENDTRY.

* gt_data 가지고 loop 돌던 머 하던 필요한 로직 작성하면 됨.

ENDFORM.

* adbc 통해서 Structure 형식으로 결과를 받는 경우

* ADBC 관련
  DATA : lcl_conn    TYPE REF TO cl_sql_connection,
         lcl_sql     TYPE REF TO cl_sql_statement,
         lcl_result  TYPE REF TO cl_sql_result_set,
         lr_data     TYPE REF TO data,
         lx_sql_exec TYPE REF TO cx_root,
         lv_sql      TYPE string,
         lv_msg      TYPE string.

  _clear gs_sflight gt_sflight.

  TRY.

      lcl_conn = cl_sql_connection=>get_connection( ).

      CREATE OBJECT lcl_sql
        EXPORTING
          con_ref = lcl_conn.

      lv_sql = | SELECT CARRID, CONNID,  FLDATE,   | &&
               |        PRICE, CURRENCY, PLANETYPE | &&
               |   FROM SFLIGHT                    | &&
               |  WHERE MANDT  = '{ sy-mandt }'    | &&
               |    AND CARRID = '{ pv_carrid }'   | &&
               |    AND CONNID = '{ pv_connid }'   | &&
               |    AND FLDATE = '{ pv_fldate }'   | .

      lcl_result = lcl_sql->execute_query( lv_sql ).

      GET REFERENCE OF gs_sflight INTO lr_data.

* Structure 형태로 받으려고 set_param_struct 메소드 호출함
      lcl_result->set_param_struct( lr_data ).

      lcl_result->next( ). "Structure 형태로 하나 받으므로 next 메소드 호출
      lcl_result->close( ).

    CATCH cx_sql_exception INTO lx_sql_exec. "에러 발생시 처리하는곳

      lv_msg = lx_sql_exec->get_text( ). "에러 메시지 받는다

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.

  ENDTRY.

* 필요한 로직 작성
