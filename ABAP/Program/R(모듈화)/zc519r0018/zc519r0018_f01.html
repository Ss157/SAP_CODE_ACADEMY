FORM get_flight .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~carrid a~currcode
         b~connid b~airpfrom b~airpto b~fltype
         c~fldate c~price
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM scarr      AS a
   INNER JOIN spfli AS b
      ON a~carrid = b~carrid
   INNER JOIN sflight AS c
      ON b~carrid = c~carrid
     AND b~connid = c~connid
   WHERE a~carrid IN so_carr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'STATUS'.
      gs_fcat-icon = 'X'.

    WHEN 'CARRID'.
      gs_fcat-just = 'C'.

    WHEN 'PRICE'.
      gs_fcat-cfieldname = 'CURRCODE'.
      gs_fcat-do_sum     = 'X'.

    WHEN 'FLTYPE'.
      gs_fcat-checkbox = 'X'.

    WHEN 'AIRPFROM'.
      gs_fcat-emphasize = 'C500'.

    WHEN 'AIRPTO'.
      gs_fcat-emphasize = 'C310'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_sort         = gt_sort
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM set_icon .

  DATA : lv_tabix TYPE sy-tabix,
         ls_color TYPE lvc_s_scol.

  CLEAR  lv_tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    gs_data-status = icon_ws_plane.

    CLEAR   : ls_color.
    REFRESH : gs_data-color.

    CASE 'X'.
      WHEN gs_data-fltype.
        ls_color-fname     = 'FLTYPE'.
        ls_color-color-col = 3.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

        APPEND ls_color TO gs_data-color.
        CLEAR  ls_color.

    ENDCASE.

    CASE gs_data-fldate.
      WHEN '20251108'.
*        ls_color-fname     = 'FLDATE'.
        ls_color-nokeycol  = 'X'.
        ls_color-color-col = 6.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

        APPEND ls_color TO gs_data-color.
        CLEAR  ls_color.

    ENDCASE.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING status color.

  ENDLOOP.

ENDFORM.
