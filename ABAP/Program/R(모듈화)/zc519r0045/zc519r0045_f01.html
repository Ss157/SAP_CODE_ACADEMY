FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_class
      it_fieldcatalog = gt_fcat.

ENDFORM.

      ENDIF.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_class INTO gs_class.

    lv_tabix = sy-tabix.

    CLEAR ls_style.
    REFRESH gs_class-style.

    ls_style-fieldname = 'CLASS'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_class-style.
    CLEAR  ls_style.

    ls_style-fieldname = 'CL_NUM'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_class-style.
    CLEAR  ls_style.

    MODIFY gt_class FROM gs_class INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

* 선택한 행의 번호를 가져오기
  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

* itab에서 선택한 행을 삭제하기
  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_class INTO gs_class INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_class TO gt_class_del.
      CLEAR  gs_class.

      DELETE gt_class INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

ENDFORM.
