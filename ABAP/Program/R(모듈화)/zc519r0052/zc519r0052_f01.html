FORM get_company .

  _clear gs_company gt_company.
  REFRESH gt_company_del.

  SELECT matnr werks ekgrp dispo beskz
         erdat erzet ernam aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_company
    FROM zc519t0035
   WHERE werks = pa_werks.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR lv_tabix.

  LOOP AT gt_company INTO gs_company.

    lv_tabix = sy-tabix.

    _clear ls_style gs_company-style.

    ls_style-fieldname = 'MATNR'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_company-style.
    CLEAR  ls_style.

    MODIFY gt_company FROM gs_company INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER lcl_event_handler=>handle_data_changed FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_company
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM handle_data_changed  USING pt_changed TYPE lvc_t_modi.

  DATA : ls_modi TYPE lvc_s_modi.

  LOOP AT pt_changed INTO ls_modi.

    READ TABLE gt_company INTO gs_company INDEX ls_modi-row_id.

    IF sy-subrc <> 0.

      CONTINUE.

    ENDIF.

    gs_company-mark = 'X'.

    MODIFY gt_company FROM gs_company INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_company INTO gs_company INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_company TO gt_company_del.
      CLEAR  gs_company.

      DELETE gt_company INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE ZC519t0035,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear  ls_save lt_save.
  CLEAR   lv_commit.
  REFRESH lt_del.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_company_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_company_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE ZC519t0035 FROM TABLE lt_del.

      IF sy-dbcnt > 0.

        lv_commit = 'X'.

      ENDIF.

    ENDIF.

  ENDIF.

  LOOP AT gt_company INTO gs_company WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_company TO ls_save.

*    IF ls_save IS INITIAL.
    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY ZC519t0035 FROM TABLE lt_save.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.
    REFRESH gt_company_del.

    PERFORM clear_mark.

    MESSAGE s002.

  ELSE. """""""""""""""

    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'W'. """""""""

  ENDIF.

ENDFORM.
