FORM get_purch .

  CLEAR   : gs_purch.
  REFRESH : gt_purch.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~menge b~meins
         c~batxt
         d~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_purch
    FROM ekko      AS a
   INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
    LEFT OUTER JOIN t161t AS c
      ON a~bsart = c~bsart
     AND a~bstyp = c~bstyp
     AND c~spras = sy-langu
    LEFT OUTER JOIN makt AS d
      ON b~matnr = d~matnr
     AND d~spras = sy-langu
   WHERE a~ebeln IN so_ebeln.

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

ENDFORM.

FORM make_fcat USING gv_key gv_field gv_text gv_ref_table gv_ref_field.

  gs_fcat-key       = gv_key.
  gs_fcat-fieldname = gv_field.
  gs_fcat-coltext   = gv_text.
  gs_fcat-ref_table = gv_ref_table.
  gs_fcat-ref_field = gv_ref_field.

  CASE gv_field.
    WHEN 'MENGE'.
      gs_fcat-qfieldname = 'MEINS'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_purch
      it_fieldcatalog = gt_fcat.

ENDFORM.
