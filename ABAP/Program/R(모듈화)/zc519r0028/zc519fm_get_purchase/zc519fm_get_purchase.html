FUNCTION zc519fm_get_purchase.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_BUKRS) TYPE  BUKRS
*"  EXPORTING
*"     REFERENCE(PE_CODE) TYPE  CHAR1
*"     REFERENCE(PE_MSG) TYPE  CHAR100
*"  TABLES
*"      PT_PURSH STRUCTURE  ZC519S0013
*"      IT_EBELN OPTIONAL
*"----------------------------------------------------------------------

*       Global data declarations

  CLEAR   : pe_code, pe_msg.
  REFRESH : pt_pursh.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~ktmng b~meins
    INTO CORRESPONDING FIELDS OF TABLE pt_pursh
    FROM ekko      AS a
   INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
   WHERE a~bukrs  = pi_bukrs
     AND a~ebeln IN it_ebeln.

  IF sy-subrc <> 0.

    pe_code = 'E'.
    pe_msg  = 'Data not found'.

  ELSE.

    pe_code = 'S'.

  ENDIF.

ENDFUNCTION.
