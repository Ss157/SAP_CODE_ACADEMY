FORM get_purchase_func .

  DATA : lt_data     TYPE zc519tt006,
         lv_code,
         lv_msg(100).

  CLEAR   : lv_code, lv_msg.
  REFRESH : lt_data, gt_purchase.

  call function 'ZC519FM_GET_PURCHASE'
    EXPORTING
      pi_bukrs = pa_bukrs
    IMPORTING
      pe_code  = lv_code
      pe_msg   = lv_msg
    TABLES
      pt_pursh = lt_data
      it_ebeln = so_ebeln[].

  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    STOP.

  ENDIF.

  MOVE-CORRESPONDING lt_data TO gt_purchase.

ENDFORM.

FORM get_purchase_met .

  DATA : lcl_purchase TYPE REF TO zc519cl002,
         lt_data      TYPE zc519tt006,
         lv_code,
         lv_msg(100).

  CLEAR   : lv_code, lv_msg.
  REFRESH : lt_data, gt_purchase.

  IF lcl_purchase IS NOT BOUND.

    CREATE OBJECT lcl_purchase.

  ENDIF.

  CALL METHOD lcl_purchase->get_purchase
    EXPORTING
      pi_bukrs = pa_bukrs
      it_ebeln = so_ebeln[]
    IMPORTING
      pe_code  = lv_code
      pe_msg   = lv_msg
    CHANGING
      pt_data  = lt_data.

  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    STOP.

  ENDIF.

  MOVE-CORRESPONDING lt_data TO gt_purchase.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_purchase_form
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_purchase_form .

  CLEAR   : gs_purchase.
  REFRESH : gt_purchase.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~ktmng b~meins
    INTO CORRESPONDING FIELDS OF TABLE gt_purchase
    FROM ekko      AS a
   INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
   WHERE a~bukrs  = pa_bukrs
     AND a~ebeln IN so_ebeln.

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_color .

  DATA : lv_tabix TYPE sy-tabix,
         ls_color TYPE lvc_s_scol.

  CLEAR : lv_tabix, ls_color.

  LOOP AT gt_purchase INTO gs_purchase.

    lv_tabix = sy-tabix.

    CLEAR   ls_color.
    REFRESH gs_purchase-color.

    IF gs_purchase-ktmng <= 5000.

      ls_color-fname     = 'KTMNG'.
      ls_color-color-col = 3.
      ls_color-color-int = 0.
      ls_color-color-inv = 0.

      APPEND ls_color TO gs_purchase-color.
      CLEAR  ls_color.

      MODIFY gt_purchase FROM gs_purchase INDEX lv_tabix
      TRANSPORTING color.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table  pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'KTMNG'.
      gs_fcat-qfieldname = 'MEINS'.
      gs_fcat-do_sum     = 'X'.

    WHEN 'MATNR'.
      gs_fcat-emphasize = 'C510'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
