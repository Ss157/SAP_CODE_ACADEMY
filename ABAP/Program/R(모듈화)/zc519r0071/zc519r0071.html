WITH +flight   AS ( SELECT carrid, connid, fldate, price, currency, planetype
                      FROM sflight
                     WHERE carrid IN ( 'AA', 'DL' )
                  ),
     +schedule AS ( SELECT carrid, connid, airpfrom
                      FROM spfli
                     WHERE carrid IN ( 'AA', 'DL' )
                  )

SELECT a~carrid, a~connid, b~fldate, b~price
  FROM +schedule AS a
 INNER JOIN +flight AS b
    ON a~carrid = b~carrid
   AND a~connid = b~connid
