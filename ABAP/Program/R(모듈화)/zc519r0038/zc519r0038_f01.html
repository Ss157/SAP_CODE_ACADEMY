FORM get_schedule .

  _clear : gs_spfli gt_spfli.

  SELECT carrid   connid airpfrom airpto
         distance distid fltype
    INTO CORRESPONDING FIELDS OF TABLE gt_spfli
    FROM spfli
   WHERE carrid IN so_carr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

*&      --> P_

FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'DISTANCE'.
      gs_fcat-qfieldname = 'DISTID'.

    WHEN 'FLTYPE'.
      gs_fcat-checkbox   = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_schedule .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER : gcl_handler->handle_double_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_spfli
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_spfli INTO gs_spfli INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

* 지금은 특정 필드를 더블클릭했을 때 팝업이 나타나는 요청이 아니고
* 행을 더블클릭하면 바로 팝업을 띄워달라는 요청을 받았음.

* get_scarr 라는 Form의 팝업 이외에도 사용이 가능하게
* 파라메터를 사용하는 Form으로 사용하는 걸 강추
  PERFORM get_scarr USING gs_spfli-carrid.

  IF gt_scarr IS INITIAL.
    MESSAGE s000 WITH 'Airline Master not found'.
    EXIT.
  ENDIF.

  CALL SCREEN '0101' STARTING AT 20 3.
ENDFORM.

FORM get_scarr  USING pv_carrid TYPE scarr-carrid.

  _clear gs_scarr gt_scarr.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr
   WHERE carrid = pv_carrid.

ENDFORM.
