FORM get_schedule .

  REFRESH : gt_schedule.

  SELECT a~carrid a~carrname a~currcode
         b~connid b~airpfrom b~airpto
    INTO CORRESPONDING FIELDS OF TABLE gt_schedule
    FROM      scarr AS a
   INNER JOIN spfli AS b
      ON a~carrid = b~carrid
   WHERE a~carrid IN so_carr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

  SORT gt_schedule BY carrid connid.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gcl_grid->set_table_for_first_display(
    EXPORTING
      i_structure_name = 'ZC519S0008'
      i_save           = 'A'
      i_default        = 'X'
      is_layout        = gs_layout
    CHANGING
      it_outtab        = gt_schedule
    ).

ENDFORM.
