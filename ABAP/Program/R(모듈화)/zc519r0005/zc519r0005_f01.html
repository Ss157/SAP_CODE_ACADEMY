FORM get_account .

  REFRESH : gt_account.

  SELECT a~bukrs a~belnr a~blart a~bktxt
         b~buzei b~bschl b~shkzg b~dmbtr b~pswsl b~kokrs
         c~bezei
         d~ltext
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM      bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
   INNER JOIN tka01 AS c
      ON b~kokrs = c~kokrs
    LEFT OUTER JOIN t003t AS d
      ON a~blart = d~blart
     AND d~spras = sy-langu
   WHERE a~bukrs  = pa_bukrs
     AND a~gjahr  = pa_gjahr
     AND a~belnr IN so_belnr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

  SORT gt_account BY bukrs belnr.

ENDFORM.

    EXPORTING

      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_structure_name = 'ZC519S0009'
      i_save           = 'A'
