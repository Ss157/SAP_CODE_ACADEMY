FORM get_airline .

  REFRESH gt_airline.

  SELECT carrier,       connect_no,   flight_date,  booking_id,
         airplane_type, total_sum,    currency_key, customer_id,
         customer_type, grade,        order_date
    INTO CORRESPONDING FIELDS OF TABLE @gt_airline
    FROM zc519cds0005
   WHERE carrier       = @pa_carr
     AND connect_no    = @pa_conn
     AND flight_date   = @pa_fldat
     AND airplane_type = @pa_plty.

  IF sy-subrc <> 0.

    MESSAGE s001 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'TOTAL_SUM'.
      gs_fcat-cfieldname = 'CURRENCY_KEY'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_airline
      it_fieldcatalog = gt_fcat.

ENDFORM.
