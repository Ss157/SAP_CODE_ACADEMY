FORM get_mat .

  CLEAR   : gs_mat.
  REFRESH : gt_mat.

  SELECT a~matnr a~mtart a~matkl
         b~werks b~pstat
         c~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mara      AS a
   INNER JOIN marc AS b
      ON a~matnr = b~matnr
    LEFT OUTER JOIN makt AS c
      ON a~matnr = c~matnr
     AND c~spras = sy-langu
   WHERE a~matnr IN so_matnr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      container_name = 'GCL_CONTAINER'.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER gcl_handler->handle_double_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_mat
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.

    EXIT.

  ENDIF.

  READ TABLE gt_mat INTO gs_mat INDEX ps_row-index.

  IF sy-subrc <> 0.

    EXIT.

  ENDIF.

  MESSAGE i000 WITH gs_mat-maktx.

ENDFORM.
