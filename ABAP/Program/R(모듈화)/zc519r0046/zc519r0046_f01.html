FORM get_idol .

  _clear gs_idol gt_idol.

  SELECT group_id    member_id     member_name
         member_addr member_gender member_phone
         erdat       erzet         ernam
         aedat       aezet         aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_idol
    FROM ZC519t0030
   WHERE group_id = pa_grp.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_idol INTO gs_idol.

    lv_tabix = sy-tabix.

    REFRESH gs_idol-style.

    ls_style-fieldname = 'GROUP_ID'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_idol-style.
    CLEAR  ls_style.

    ls_style-fieldname = 'MEMBER_ID'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_idol-style.
    CLEAR  ls_style.

    MODIFY gt_idol FROM gs_idol INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM display_alv .

    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_idol
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_idol INTO gs_idol INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_idol TO gt_idol_del.
      CLEAR  gs_idol.

      DELETE gt_idol INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE ZC519t0030,
         lt_save LIKE TABLE OF ls_save,
         lt_del  LIKE TABLE OF ls_save.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_idol_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_idol_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE ZC519t0030 FROM TABLE lt_del.

    ENDIF.

  ENDIF.

  LOOP AT gt_idol INTO gs_idol.

    MOVE-CORRESPONDING gs_idol TO ls_save.

    IF ls_save-erdat IS INITIAL.
      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.
    ELSE.
      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.
    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY ZC519t0030 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.
      REFRESH gt_idol_del.

      MESSAGE s002.
    ENDIF.

  ENDIF.

ENDFORM.
