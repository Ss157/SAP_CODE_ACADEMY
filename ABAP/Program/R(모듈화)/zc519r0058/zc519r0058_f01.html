FORM get_account .

  _clear gs_account gt_account.

  SELECT a~blart a~budat
         b~belnr b~buzei b~bschl b~koart b~gsber
         b~hkont b~shkzg b~dmbtr b~pswsl
         c~ltext
         d~gtext
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM bkpf      AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart = c~blart
     AND c~spras = sy-langu
    LEFT OUTER JOIN tgsbt AS d
      ON b~gsber = d~gsber
     AND d~spras = sy-langu
   WHERE a~bukrs = pa_bukrs
     AND a~gjahr = pa_gjahr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER : "gcl_handler->handle_hotspot      FOR gcl_grid,
                gcl_handler->on_button_click     FOR gcl_grid,
                gcl_handler->handle_double_click FOR gcl_grid,
                gcl_handler->handle_toolbar      FOR gcl_grid,
                gcl_handler->handle_user_command FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_account
      it_fieldcatalog = gt_fcat.

ENDFORM.

*&      --> E_COLUMN_ID
*&      --> E_ROW_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_col TYPE lvc_s_col
                           ps_row TYPE lvc_s_row.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_account INTO gs_account INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'BELNR'.
      SET PARAMETER ID : 'BLN' FIELD gs_account-belnr,
                         'BUK' FIELD pa_bukrs,
                         'GJR' FIELD pa_gjahr.

      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR  lv_tabix.

  LOOP AT gt_account INTO gs_account WHERE buzei = '0001'.

    lv_tabix = sy-tabix.

    _clear ls_style gs_account-style.

    ls_style-fieldname = 'BELNR'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_button.

    APPEND ls_style TO gs_account-style.
    CLEAR  ls_style.

    MODIFY gt_account FROM gs_account INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.
