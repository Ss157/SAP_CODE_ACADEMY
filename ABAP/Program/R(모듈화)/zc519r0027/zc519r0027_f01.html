FORM get_order .

  DATA : lt_data     TYPE zc519tt005,
         lcl_order   TYPE REF TO zc519cl002,
         lv_msg(100),
         lv_code.

  CLEAR   : lv_msg, lv_code.
  REFRESH : lt_data, gt_result.

  IF lcl_order IS NOT BOUND.

    CREATE OBJECT lcl_order.

  ENDIF.

  CALL METHOD lcl_order->get_product_order
    EXPORTING
      iv_plnbez = pa_plnbe
      iv_pwerk  = pa_pwerk
    IMPORTING
      ev_msg    = lv_msg
      ev_code   = lv_code
      et_prod   = lt_data.

  IF lv_code = 'E'.

    MESSAGE s001.
    STOP.

  ENDIF.

  MOVE-CORRESPONDING lt_data TO gt_result.

ENDFORM.

FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PSMNG'.
      gs_fcat-emphasize = 'C300'.
      gs_fcat-qfieldname = 'AMEIN'.

    WHEN 'AMEIN'.
      gs_fcat-emphasize = 'C300'.

    WHEN 'GAMNG'.
      gs_fcat-qfieldname = 'GMEIN'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM get_color .

  DATA : ls_color TYPE lvc_s_scol,
         lv_tabix TYPE sy-tabix.

  CLEAR : ls_color, lv_tabix.

  LOOP AT gt_result INTO gs_result.

    lv_tabix = sy-tabix.

    CLEAR   : ls_color.
    REFRESH : gs_result-color.

    IF gs_result-psmng <= 10000.

      ls_color-fname     = 'PSMNG'.
      ls_color-color-col = 6.
      ls_color-color-int = 1.
      ls_color-color-inv = 0.

      APPEND ls_color TO gs_result-color.
      CLEAR  ls_color.

      MODIFY gt_result FROM gs_result INDEX lv_tabix
      TRANSPORTING color.

      CLEAR : gs_result.

    ENDIF.

  ENDLOOP.

ENDFORM.
