FUNCTION zc519fm_get_bill.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_WERKS) TYPE  WERKS_D
*"  EXPORTING
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"  TABLES
*"      ET_DATA STRUCTURE  ZC519T0040
*"----------------------------------------------------------------------

*       Global data declarations

  CLEAR   : ev_msg, ev_code.
  REFRESH : et_data.

  SELECT matnr werks stlan stlnr
         stlal losbs losvn bmein
         erdat erzet ernam
         aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE et_data
    FROM zc519t0040
   WHERE werks = iv_werks
   ORDER BY matnr werks stlan stlnr stlal.

  IF sy-subrc <> 0.

    ev_code = 'E'.
    ev_msg  = 'Data not found'.

  ELSE.

    ev_code = 'S'.

  ENDIF.

ENDFUNCTION.
