FORM get_purchase .

  DATA : lv_code,
         lv_msg(100).

  CLEAR   : gs_purchase, lv_code, lv_msg.
  REFRESH : gt_purchase.

  call function 'ZC519FM_GET_PURCHASE2'
    IMPORTING
      ev_code     = lv_code
      ev_msg      = lv_msg
    TABLES
      et_purchase = gt_purchase
      it_ebeln    = so_ebeln.

  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'KTMNG'.
      gs_fcat-qfieldname = 'MEINS'.
      gs_fcat-do_sum     = 'X'.

    WHEN 'EBELN'.
      gs_fcat-hotspot    = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid,
                lcl_event_handler=>handle_hotspot      FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_purchase
      it_fieldcatalog = gt_fcat
      it_sort         = gt_sort.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_purchase INTO gs_purchase INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.
      SET PARAMETER ID 'MAT' FIELD gs_purchase-matnr.

      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.

*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_purchase INTO gs_purchase INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'EBELN'.
      CALL FUNCTION 'ME_DISPLAY_PURCHASE_DOCUMENT'
        EXPORTING
          i_ebeln = gs_purchase-ebeln.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.
