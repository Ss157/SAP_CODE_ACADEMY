FORM get_mat .

  CLEAR   : gs_mat.
  REFRESH : gt_mat.

  SELECT a~matnr a~mtart a~matkl a~mbrsh
         b~werks
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mara      AS a
   INNER JOIN marc AS b
      ON a~matnr = b~matnr
   WHERE b~werks  = pa_werks
     AND a~matnr IN so_matnr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM get_text .

  CLEAR   : gs_t134t, gs_t023t, gs_t137t.
  REFRESH : gt_t134t, gt_t023t, gt_t137t.

  SELECT mtart mtbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t134t
    FROM t134t
   WHERE spras = sy-langu.

  SELECT matkl wgbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t023t
    FROM t023t
   WHERE spras = sy-langu.

  SELECT mbrsh mbbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t137t
    FROM t137t
   WHERE spras = sy-langu.

ENDFORM.

FORM modify_mat .

  DATA : lv_tabix TYPE sy-tabix.
  CLEAR  lv_tabix.

  LOOP AT gt_mat INTO gs_mat.

    lv_tabix = sy-tabix.

    READ TABLE gt_t134t INTO gs_t134t WITH KEY mtart = gs_mat-mtart.

    IF sy-subrc = 0.

      gs_mat-mtbez = gs_t134t-mtbez.

      MODIFY gt_mat FROM gs_mat INDEX lv_tabix
      TRANSPORTING mtbez.

    ENDIF.

    READ TABLE gt_t023t INTO gs_t023t WITH KEY matkl = gs_mat-matkl.

    IF sy-subrc = 0.

      gs_mat-wgbez = gs_t023t-wgbez.

      MODIFY gt_mat FROM gs_mat INDEX lv_tabix
      TRANSPORTING wgbez.

    ENDIF.

    READ TABLE gt_t137t INTO gs_t137t WITH KEY mbrsh = gs_mat-mbrsh.

    IF sy-subrc = 0.

      gs_mat-mbbez = gs_t137t-mbbez.

      MODIFY gt_mat FROM gs_mat INDEX lv_tabix
      TRANSPORTING mbbez.

    ENDIF.

  ENDLOOP.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra      = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-row = 'X'.
  gs_stable-col = 'X'.

          'X' 'WERKS' ' ' 'MARC'  ' ',
          ' ' 'MTART' ' ' 'MARA'  ' ',
          ' ' 'MTBEZ' ' ' 'T134T' ' ',
          ' ' 'MATKL' ' ' 'MARA'  ' ',
          ' ' 'WGBEZ' ' ' 'T023T' ' ',
          ' ' 'MBRSH' ' ' 'MARA'  ' ',
          ' ' 'MBBEZ' ' ' 'T137T' ' '.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_mat
      it_fieldcatalog = gt_fcat.

ENDFORM.
