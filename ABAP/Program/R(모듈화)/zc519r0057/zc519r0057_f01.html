FORM check_param .

  DATA : lv_werks TYPE t001w-werks.

  CLEAR  lv_werks.

  IF pa_werks IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

    INTO lv_werks
    FROM t001w
   WHERE werks = pa_werks.

FORM get_mat .

  _clear  gs_mat gt_mat.
  REFRESH gt_mat_del.

  SELECT a~werks a~mtart a~flref a~envop
         a~kkref a~vmtpo a~menge a~meins
         a~erdat a~erzet a~ernam
         a~aedat a~aezet a~aenam
         b~mtbez
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM zc519t0037       AS a
    LEFT OUTER JOIN t134t AS b
      ON a~mtart = b~mtart
     AND b~spras = sy-langu
   WHERE a~werks = pa_werks.

ENDFORM.

  CREATE OBJECT gcl_grid
    EXPORTING

  ENDIF.

  SET HANDLER : gcl_handler->handle_changed_finished FOR gcl_grid,
                gcl_handler->handle_toolbar          FOR gcl_grid,
                gcl_handler->handle_user_command     FOR gcl_grid.

* ALV가 뭔가 수정하고 있음을 감지시킴

    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_mat
      it_fieldcatalog      = gt_fcat.

FORM handle_data_changed  USING pcl_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : ls_modi  TYPE lvc_s_modi,
         lv_mtbez TYPE t134t-mtbez.

  CLEAR : ls_modi, lv_mtbez.

  LOOP AT pcl_changed->mt_mod_cells INTO ls_modi.

    READ TABLE gt_mat INTO gs_mat INDEX ls_modi-row_id.

    IF sy-subrc <> 0.

      CONTINUE.

    ENDIF.

    CASE ls_modi-fieldname.
      WHEN 'MTART'.

        SELECT SINGLE mtbez
          INTO lv_mtbez
          FROM t134t
         WHERE mtart = ls_modi-value
           AND spras = sy-langu.

        CALL METHOD pcl_changed->modify_cell
          EXPORTING
            i_fieldname = 'MTBEZ'
            i_row_id    = ls_modi-row_id
            i_value     = lv_mtbez.

*        PERFORM modify_cell USING pcl_changed

    ENDCASE.

    gs_mat-mark = 'X'.

    MODIFY gt_mat FROM gs_mat INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_mat INTO gs_mat INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_mat TO gt_mat_del.
      CLEAR  gs_mat.

      DELETE gt_mat INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE zc519t0037,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear  ls_save lt_save.
  CLEAR   lv_commit.
  REFRESH lt_del.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_mat_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_mat_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE zc519t0037 FROM TABLE lt_del.

      IF sy-dbcnt > 0.

        lv_commit = 'X'.

      ENDIF.

    ENDIF.

  ENDIF.

  LOOP AT gt_mat INTO gs_mat WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_mat TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY zc519t0037 FROM TABLE lt_save.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.
    REFRESH gt_mat_del.

    PERFORM reset_mark.

    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH TEXT-f04 DISPLAY LIKE 'W'.

  ENDIF.

ENDFORM.

FORM handle_changed_finished  USING pv_modified
                                    pt_modi TYPE lvc_t_modi.

  DATA : ls_modi TYPE lvc_s_modi.

*  IF pv_modified IS INITIAL.
*    EXIT.
*  ENDIF.

  CHECK pv_modified IS NOT INITIAL. " 위에 꺼랑 같은 거
  " 이 조건을 만족하면 통과한다.

  LOOP AT pt_modi INTO ls_modi.

    READ TABLE gt_mat INTO gs_mat INDEX ls_modi-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    gs_mat-mark = 'X'.

    CASE ls_modi-fieldname.
      WHEN 'MTART'.

        SELECT SINGLE mtbez
          INTO gs_mat-mtbez "여기는 인터널 테이블이 다 끝난 상황
          FROM t134t
         WHERE mtart = gs_mat-mtart
           AND spras = sy-langu.

    ENDCASE.

    MODIFY gt_mat FROM gs_mat INDEX ls_modi-row_id
    TRANSPORTING mark mtbez.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM handle_user_command  USING pv_ucomm.

  CASE pv_ucomm.
    WHEN 'CREATE'.
      PERFORM create_row.
      PERFORM refresh_grid.

    WHEN 'DELETE'.

    WHEN 'TOGGLE'.

  ENDCASE.
