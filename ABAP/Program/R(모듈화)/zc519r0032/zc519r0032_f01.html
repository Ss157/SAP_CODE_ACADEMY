FORM get_order .

  DATA : lcl_order   TYPE REF TO zc519cl002,
         lv_code,
         lv_msg(100).

  CLEAR   : gs_order, lv_code, lv_msg.
  REFRESH : gt_order.

  IF  lcl_order IS NOT BOUND.

    CREATE OBJECT lcl_order.

  ENDIF.

  CALL METHOD lcl_order->get_order
    EXPORTING
      it_plnbez = so_plnbe[]
      iv_pwerk  = pa_pwerk
    IMPORTING
      ev_code   = lv_code
      ev_msg    = lv_msg
    CHANGING
      et_order  = gt_order.

  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PSMNG'.
      gs_fcat-qfieldname = 'AMEIN'.

    WHEN 'GAMNG'.
      gs_fcat-qfieldname = 'GMEIN'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_order IS NOT BOUND.

    CREATE OBJECT gcl_order.

  ENDIF.

  SET HANDLER gcl_order->handle_double_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_order
      it_fieldcatalog = gt_fcat.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.

    EXIT.

  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'AUFNR'.
      PERFORM show_plnbez_txt USING ps_row.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*&      --> PS_ROW
*&---------------------------------------------------------------------*
FORM show_plnbez_txt  USING ps_row TYPE lvc_s_row.

  READ TABLE gt_order INTO gs_order INDEX ps_row-index.

  IF sy-subrc <> 0.

    EXIT.

  ENDIF.

  MESSAGE i000 WITH gs_order-plnbez_txt.

ENDFORM.
