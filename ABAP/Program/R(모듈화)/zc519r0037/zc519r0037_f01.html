FORM get_mat_bom .

  DATA : lcl_mat_bom TYPE REF TO zc519cl003,
         lt_bill     TYPE TABLE OF zc519s0015,
         lv_code,
         lv_msg(100).

  CLEAR   : gs_bill, lcl_mat_bom, lv_code, lv_msg.
  REFRESH : gt_bill, lt_bill.

  IF lcl_mat_bom IS NOT BOUND.

    CREATE OBJECT lcl_mat_bom.

  ENDIF.

  CALL METHOD lcl_mat_bom->get_mat_bom
    EXPORTING
      iv_werks = pa_werks
      it_matnr = so_matnr[]
    IMPORTING
      ev_code  = lv_code
      ev_msg   = lv_msg
    CHANGING
      et_data  = lt_bill.

  IF lv_code = 'E'.

    MESSAGE s000 WITH lv_msg.
    LEAVE LIST-PROCESSING.

  ENDIF.

  MOVE-CORRESPONDING lt_bill TO gt_bill.

ENDFORM.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.

    WHEN 'SANKA'.
      gs_fcat-checkbox   = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid,
                lcl_event_handler=>handle_hotspot      FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_bill
      it_fieldcatalog = gt_fcat
      it_sort         = gt_sort.

ENDFORM.

FORM set_icon_color .

  DATA : ls_color TYPE lvc_s_scol,
         lv_tabix TYPE sy-tabix.

  CLEAR : lv_tabix, ls_color.

  LOOP AT gt_bill INTO gs_bill.

    lv_tabix = sy-tabix.

    gs_bill-status = icon_material.

    CLEAR   : ls_color.
    REFRESH : gs_bill-color. """"

    IF gs_bill-menge <= 10.

      ls_color-fname     = 'MENGE'.
      ls_color-color-col = 3.
      ls_color-color-int = 0.
      ls_color-color-inv = 0.

      APPEND ls_color TO gs_bill-color.
      CLEAR  ls_color.

    ENDIF.

    MODIFY gt_bill FROM gs_bill INDEX lv_tabix
    TRANSPORTING status color.

    CLEAR : gs_bill.

  ENDLOOP.

ENDFORM.

*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_bill INTO gs_bill INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'STLNR'.
      SET PARAMETER ID : 'MAT' FIELD gs_bill-matnr,
                         'CSV' FIELD gs_bill-stlan,
                         'WRK' FIELD gs_bill-werks.

      CALL TRANSACTION 'CS03' AND SKIP FIRST SCREEN.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.

*& Form handle_hotspot
*&---------------------------------------------------------------------*

*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_bill INTO gs_bill INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'IDNRK'.
      SET PARAMETER ID 'MAT' FIELD gs_bill-idnrk.

      CALL TRANSACTION 'MM03' AND SKIP FIRST SCREEN.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.
