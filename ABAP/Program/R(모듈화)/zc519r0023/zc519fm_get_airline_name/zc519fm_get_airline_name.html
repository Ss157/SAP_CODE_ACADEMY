FUNCTION zc519fm_get_airline_name.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_CARRID) TYPE  S_CARR_ID
*"  EXPORTING
*"     REFERENCE(PE_CARRNAME) TYPE  S_CARRNAME
*"     REFERENCE(PE_CODE) TYPE  C
*"     REFERENCE(PE_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE carrname
    INTO pe_carrname
    FROM scarr
   WHERE carrid = pi_carrid.

  IF sy-subrc <> 0.

    pe_code = 'E'.
    pe_msg  = 'Data Not Found'.

  ELSE.

    pe_code = 'S'.
    " 성공했을 때 msg는 안 써도 된다.

  ENDIF.

ENDFUNCTION.
