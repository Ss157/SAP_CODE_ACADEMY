FORM get_airline .

  CLEAR   : gs_airline.
  REFRESH : gt_airline.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_airline
    FROM scarr
   WHERE carrid IN so_carr.

  IF sy-subrc <> 0.
    MESSAGE s001.
*    STOP.
    LEAVE LIST-PROCESSING. "= stop과 같은 의미
  ENDIF.

ENDFORM.

FORM modify_airline .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_airline INTO gs_airline.

    lv_tabix = sy-tabix.

    CASE gs_airline-carrid.
      WHEN 'JL'.
        gs_airline-carrid   = 'KA'.
        gs_airline-carrname = 'Korean Air'.
        gs_airline-currcode = 'KRW'.
        gs_airline-url      = 'www.korean-air.ck.kr'.

        MODIFY gt_airline FROM gs_airline INDEX lv_tabix
        TRANSPORTING carrid carrname currcode url.

    ENDCASE.

  ENDLOOP.

ENDFORM.
