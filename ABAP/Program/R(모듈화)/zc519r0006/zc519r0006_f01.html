FORM get_idol_info .

  REFRESH : gt_idol.

  SELECT a~group_id    a~group_txt     b~member_id
         b~member_name b~member_gender a~group_url
    INTO CORRESPONDING FIELDS OF TABLE gt_idol
    FROM zc519t0029      AS a
   INNER JOIN zc519t0030 AS b
      ON a~group_id = b~group_id
   WHERE a~group_id IN so_grpid.

  IF sy-subrc <> 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_structure_name = 'ZC519S0010'
      i_save           = 'A'
      i_default        = 'X'
      is_layout        = gs_layout
    CHANGING
      it_outtab        = gt_idol.

ENDFORM.
