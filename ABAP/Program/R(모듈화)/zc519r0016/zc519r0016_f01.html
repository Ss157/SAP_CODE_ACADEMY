FORM get_order .

  CLEAR   : gs_order, gs_order2.
  REFRESH : gt_order, gt_order2.

  SELECT a~aufnr a~gstrp a~gltrp a~plnbez a~gamng a~gmein
         b~posnr b~matnr b~psmng b~amein  b~bwtar
         c~maktx AS plnbez_txt
         d~maktx AS matnr_txt
    INTO CORRESPONDING FIELDS OF TABLE gt_order
    FROM afko      AS a
   INNER JOIN afpo AS b
      ON a~aufnr = b~aufnr
    LEFT OUTER JOIN makt AS c
      ON a~plnbez = c~matnr
     AND c~spras = sy-langu
    LEFT OUTER JOIN makt AS d
      ON b~matnr = d~matnr
     AND d~spras = sy-langu
   WHERE a~plnbez IN so_plnbe
     AND b~pwerk   = pa_pwerk.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM move_order .

  LOOP AT gt_order INTO gs_order.

    CASE gs_order-bwtar.
      WHEN '02'.

        MOVE-CORRESPONDING gs_order TO gs_order2.
        APPEND gs_order2 TO gt_order2.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra      = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-row = 'X'.
  gs_stable-col = 'X'.

  PERFORM make_fcat USING :
          'X' 'AUFNR'      ' ' 'AFKO' '     ',
          'X' 'POSNR'      ' ' 'AFPO' '     ',
          ' ' 'PLNBEZ'     ' ' 'AFKO' '     ',
          ' ' 'PLNBEZ_TXT' ' ' 'MAKT' 'MAKTX',
          ' ' 'MATNR'      ' ' 'AFPO' '     ',
          ' ' 'MATNR_TXT ' ' ' 'MAKT' 'MAKTX',
          ' ' 'PSMNG'      ' ' 'AFPO' '     ',
          ' ' 'AMEIN'      ' ' 'AFPO' '     ',
          ' ' 'BWTAR'      ' ' 'AFPO' '     '.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PSMNG'.
      gs_fcat-qfieldname = 'AMEIN'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_order
      it_fieldcatalog = gt_fcat.

ENDFORM.
