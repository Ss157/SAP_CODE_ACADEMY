FORM display_dynamic_list .

  DATA : lt_mara    TYPE TABLE OF mara,
         lt_marc    TYPE TABLE OF marc,
         lt_sflight TYPE TABLE OF sflight.

  FIELD-SYMBOLS : <lft_itab> TYPE ANY TABLE.

  REFRESH  : lt_mara, lt_marc, lt_sflight.
  UNASSIGN : <lft_itab>.

  CASE pa_tab.
    WHEN 'MARA'.
      ASSIGN lt_mara TO <lft_itab>.

    WHEN 'MARC'.
      ASSIGN lt_marc TO <lft_itab>.

    WHEN 'SFLIGHT'.
      ASSIGN lt_sflight TO <lft_itab>.

  ENDCASE.

  IF <lft_itab> IS NOT ASSIGNED.

    MESSAGE s000 WITH 'Table name is wrong'
    DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE <lft_itab>
    FROM (pa_tab).

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

  cl_demo_output=>display( <lft_itab> ).

ENDFORM.

FORM write_field_dynamic_itab .

  DATA : lt_mara    TYPE TABLE OF mara,
         lt_marc    TYPE TABLE OF marc,
         lt_sflight TYPE TABLE OF sflight,
         lv_tabname TYPE dd03l-tabname.

  FIELD-SYMBOLS : <lft_itab>  TYPE ANY TABLE,
                  <lfs_itab>  TYPE any,
                  <lfv_field> TYPE any.

  CLEAR    : lv_tabname.
  REFRESH  : lt_mara, lt_marc, lt_sflight.
  UNASSIGN : <lft_itab>, <lfs_itab>, <lfv_field>.

  CASE pa_tab.
    WHEN 'MARA'.
      lv_tabname = 'LT_MARA'.

    WHEN 'MARC'.
      lv_tabname = 'LT_MARC'.

    WHEN 'SFLIGHT'.
      lv_tabname = 'LT_SFLIGHT'.

  ENDCASE.

  ASSIGN (lv_tabname) TO <lft_itab>.

  IF <lft_itab> IS NOT ASSIGNED.

    MESSAGE s000 WITH 'Table name is wrong'
    DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE <lft_itab>
    FROM (pa_tab).

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

* 필드를 뽑기
  LOOP AT <lft_itab> ASSIGNING <lfs_itab>.

    "UNASSIGN랑 IS ASSIGNED을 같이 써줘서
    "필드 심볼이 잘못될 것을 막아줘야 한다.

    UNASSIGN <lfv_field>. " assign전에 unassign을 반드시 하기
    ASSIGN COMPONENT 1 OF STRUCTURE <lfs_itab> TO <lfv_field> .

    IF <lfv_field> IS ASSIGNED.
      WRITE : <lfv_field>. " 첫 번째 필드는 mandt일 것임.
    ENDIF.

    UNASSIGN <lfv_field>.
    ASSIGN COMPONENT 2 OF STRUCTURE <lfs_itab> TO <lfv_field> .

    IF <lfv_field> IS ASSIGNED.
      WRITE : <lfv_field>.
    ENDIF.

    UNASSIGN <lfv_field>.
    ASSIGN COMPONENT 3 OF STRUCTURE <lfs_itab> TO <lfv_field> .

    IF <lfv_field> IS ASSIGNED.
      WRITE : <lfv_field>.
    ENDIF.

    UNASSIGN <lfv_field>.
    ASSIGN COMPONENT 4 OF STRUCTURE <lfs_itab> TO <lfv_field> .

    IF <lfv_field> IS ASSIGNED.
      WRITE : <lfv_field>,/.
    ENDIF.

  ENDLOOP.

ENDFORM.

FORM write_field_dynamic_itab_2 .

  DATA : lt_sflight TYPE TABLE OF sflight,
         lv_field   TYPE fieldname.

  FIELD-SYMBOLS : <lft_itab> TYPE ANY TABLE,
                  <lfs_itab> TYPE any,
                  <lfv_itab> TYPE any.

  REFRESH  : lt_sflight.
  UNASSIGN : <lft_itab>, <lfs_itab>, <lfv_itab>.

  ASSIGN ('LT_SFLIGHT') TO <lft_itab>.

* ASSIGNED 체크 필수

  lv_field = '*'.

  SELECT (lv_field)
    INTO CORRESPONDING FIELDS OF TABLE <lft_itab>
    FROM sflight.

* sy-subrc 체크

*  lv_field = 'CARRID'.
**  READ TABLE <lft_itab> ASSIGNING <lfs_itab> WITH KEY ('CARRID') = 'AA'.
*  READ TABLE <lft_itab> ASSIGNING <lfs_itab> WITH KEY (lv_field) = 'AA'.

  LOOP AT <lft_itab> ASSIGNING <lfs_itab>.

    UNASSIGN <lfv_itab>.
    ASSIGN COMPONENT 'CARRID' OF STRUCTURE <lfs_itab> TO <lfv_itab>.

    IF <lfv_itab> IS ASSIGNED.

      CASE <lfv_itab>.
        WHEN 'JL'.
          <lfv_itab> = 'KA'.

      ENDCASE.

    ENDIF.

  ENDLOOP.

ENDFORM.

  DATA : lt_fagl   TYPE TABLE OF faglflext, "얘가 cds 뷰라서 워닝 뜸
         "data definition 쓰라고 워닝 뜨는 거임
         ls_fagl   LIKE LINE OF lt_fagl,

* HSL01 ~ HSL12 필드에 10,000을 더하는 로직
  LOOP AT lt_fagl INTO ls_fagl.

*  구성해야 할 필드명은 'HSL' + 01, 02, ~ 12
    DO 12 TIMES.

      lv_mon = sy-index.

      CLEAR    lv_field.
      UNASSIGN <lfv_field>.

* ASSIGN 구문을 사용할때는 할당하고자 하는 변수(스트럭쳐,ITAB등) 의
* FULL NAME 을 줘야 함
      CONCATENATE 'LS_FAGL-HSL' lv_mon INTO lv_field.
      ASSIGN (lv_field) TO <lfv_field>.

* ASSIGN COMPONENT : STRUCTURE 에만 사용 가능
*      CONCATENATE 'HSL' lv_mon INTO lv_field.
*      ASSIGN COMPONENT lv_field OF STRUCTURE ls_fagl TO <lfv_field>.

      IF <lfv_field> IS ASSIGNED.
        <lfv_field> = <lfv_field> + 10000.
      ENDIF.

    ENDDO.

  ENDLOOP.

ENDFORM.
