FORM get_airline .

  CLEAR   : gs_airline.
  REFRESH : gt_airline.

  SELECT a~carrid     AS air_carrier    a~connid    AS connection
         a~fldate     AS depature_date  a~price     AS cost_amount
         a~paymentsum AS pay_total      a~currency  AS curr_key
         b~bookid     AS booking_id     b~customid  AS customer
         b~custtype   AS customer_type  b~agencynum AS agency_number
    INTO CORRESPONDING FIELDS OF TABLE gt_airline
    FROM sflight    AS a
   INNER JOIN sbook AS b
      ON a~carrid = b~carrid
     AND a~connid = b~connid
     AND a~fldate = b~fldate
   WHERE a~carrid  = pa_carr
     AND a~connid IN so_conn
     AND a~fldate IN so_date
     AND b~bookid IN so_bkid.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'COST_AMOUNT' OR 'PAY_TOTAL'.
      gs_fcat-cfieldname = 'CURR_KEY'.
      gs_fcat-do_sum     = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_airline
      it_fieldcatalog = gt_fcat
      it_sort         = gt_sort.

ENDFORM.
