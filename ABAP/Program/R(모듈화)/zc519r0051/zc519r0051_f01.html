FORM check_param .

  DATA : lv_class TYPE zc531tclass-class.

  CLEAR  lv_class.

  IF pa_class IS INITIAL.

    MESSAGE s000 WITH 'Input Class Number' DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

  SELECT SINGLE class
    INTO lv_class
    FROM zc531tclass
   WHERE class = pa_class.

  IF sy-subrc <> 0.

    MESSAGE s000 WITH 'Input is not validation' DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR lv_tabix.

  LOOP AT gt_class INTO gs_class.

    lv_tabix = sy-tabix.

    _clear ls_style gs_class-style.

    ls_style-fieldname = 'CL_NUM'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_class-style.
    CLEAR  ls_style.

    MODIFY gt_class FROM gs_class INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER lcl_event_handler=>handle_data_changed FOR gcl_grid.

  " ALV가 뭔가 수정하고 있음을 감지시키기
  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_class
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM handle_data_changed  USING pt_changed TYPE lvc_t_modi.

  DATA : ls_modi TYPE lvc_s_modi.

  LOOP AT pt_changed INTO ls_modi.

    READ TABLE gt_class INTO gs_class INDEX ls_modi-row_id.

    IF sy-subrc <> 0.

      CONTINUE.

    ENDIF.

    gs_class-mark = 'X'.

    MODIFY gt_class FROM gs_class INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.

FORM delete_row .

  DATA : lv_cnt TYPE i,
         ls_del LIKE LINE OF gt_class_del.

  _clear gs_row gt_row.
  CLEAR  lv_cnt.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  lv_cnt = lines( gt_row ).


*  IF gt_row IS INITIAL.
  IF lv_cnt < 1.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_class INTO gs_class INDEX gs_row-index.

    IF sy-subrc = 0.

      MOVE-CORRESPONDING gs_class TO ls_del.

      APPEND ls_del TO gt_class_del.
      CLEAR  ls_del.

      DELETE gt_class INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE ZC519t0002,
         lt_save   LIKE TABLE OF ls_save,
         lv_commit.

  _clear ls_save lt_save.
  CLEAR  lv_commit.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_class_del IS NOT INITIAL.

    DELETE ZC519t0002 FROM TABLE gt_class_del.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  LOOP AT gt_class INTO gs_class WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_class TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum. "시스템 현재 날짜
*     ls_save-erdat = sy-datlo."현지 날짜
      ls_save-erzet = sy-uzeit. "시스템 현재시간
*     ls_save-erzet = sy-timlo."현지 시간
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum. "시스템 현재 날짜
*     ls_save-aedat = sy-datlo."현지 날짜
      ls_save-aezet = sy-uzeit. "시스템 현재시간
*     ls_save-aezet = sy-timlo."현지 시간
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL. " lt_save가 반드시 값이 있을 거라고 생각 X

    MODIFY ZC519t0002 FROM TABLE lt_save.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.
    REFRESH gt_class_del.

    PERFORM clear_mark.

    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'W'.

  ENDIF.

ENDFORM.

FORM clear_mark .

  DATA : lv_tabix TYPE sy-tabix.

  CLEAR  lv_tabix.

  LOOP AT gt_class INTO gs_class WHERE mark = 'X'.

    lv_tabix = sy-tabix.

    CLEAR gs_class-mark.

    MODIFY gt_class FROM gs_class INDEX lv_tabix
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.
