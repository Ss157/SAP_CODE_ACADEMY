FORM get_cost .

  _clear  gs_cost gt_cost.
  REFRESH gt_cost_del.

  SELECT a~kokrs a~kostl a~bukrs a~kosar verak wrbtr waers bkzer
         a~erdat a~erzet a~ernam a~aedat aezet aenam
         b~ktext
    INTO CORRESPONDING FIELDS OF TABLE gt_cost
    FROM ZC519t0036      AS a
    LEFT OUTER JOIN cskt AS b
      ON a~kokrs = b~kokrs
     AND a~kostl = b~kostl
     AND b~spras = sy-langu
   WHERE a~kokrs = pa_kokrs.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR lv_tabix.

  LOOP AT gt_cost INTO gs_cost.

    lv_tabix = sy-tabix.

    _clear ls_style gs_cost-style.

    ls_style-fieldname = 'KOSTL'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_cost-style.
    CLEAR  ls_style.

    MODIFY gt_cost FROM gs_cost INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_data_changed FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_cost
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_cost INTO gs_cost INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_cost TO gt_cost_del.
      CLEAR  gs_cost.

      DELETE gt_cost INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE ZC519t0036,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear  ls_save lt_save.
  CLEAR   lv_commit.
  REFRESH lt_del.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_cost_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_cost_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE ZC519t0036 FROM TABLE lt_del.

      IF sy-dbcnt > 0.

        lv_commit = 'X'.

      ENDIF.

    ENDIF.

  ENDIF.

  LOOP AT gt_cost INTO gs_cost WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_cost TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY ZC519t0036 FROM TABLE lt_save.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  IF lv_commit = 'X'.

    COMMIT WORK AND WAIT.

    REFRESH gt_cost_del.

    PERFORM reset_mark.

    MESSAGE s002.

  ENDIF.

ENDFORM.

*& Form handle_data_changed
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> ER_DATA_CHANGED_>MT_MOD_CELLS
*&---------------------------------------------------------------------*
FORM handle_data_changed  USING pcl_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : ls_modi  TYPE lvc_s_modi,
         lv_ktext TYPE cskt-ktext.

  CLEAR : ls_modi, lv_ktext.

  LOOP AT pcl_changed->mt_mod_cells INTO ls_modi.

    READ TABLE gt_cost INTO gs_cost INDEX ls_modi-row_id.

    IF sy-subrc <> 0.

      CONTINUE.

    ENDIF.

    CASE ls_modi-fieldname.
      WHEN 'KOSTL'.

        SELECT SINGLE ktext
          INTO lv_ktext
          FROM cskt
         WHERE kostl = ls_modi-value
           AND kokrs = gs_cost-kokrs
           AND spras = sy-langu.

        CALL METHOD pcl_changed->modify_cell
          EXPORTING
            i_fieldname = 'KTEXT'
            i_row_id    = ls_modi-row_id
            i_value     = lv_ktext.

    ENDCASE.

    gs_cost-mark = 'X'.

    MODIFY gt_cost FROM gs_cost INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.

FORM check_param .

  DATA : lv_kokrs TYPE tka01-kokrs.

  CLEAR  lv_kokrs.

  IF pa_kokrs IS INITIAL.

    MESSAGE s000 WITH TEXT-f02 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

  SELECT SINGLE kokrs
    INTO lv_kokrs
    FROM tka01
   WHERE kokrs = pa_kokrs.

  IF sy-subrc <> 0.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.
