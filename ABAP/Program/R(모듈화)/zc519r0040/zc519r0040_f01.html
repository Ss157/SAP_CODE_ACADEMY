*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_mat .

  _clear gs_mat gt_mat.

  SELECT a~matnr a~vpsta a~mbrsh a~meins
         b~werks b~ekgrp b~dismm b~beskz
         c~eknam
         d~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mara      AS a
   INNER JOIN marc AS b
      ON a~matnr = b~matnr
    LEFT OUTER JOIN t024 AS c
      ON b~ekgrp = c~ekgrp
    LEFT OUTER JOIN makt AS d
      ON a~matnr = d~matnr
     AND d~spras = sy-langu
   WHERE b~werks = pa_werks.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'EKGRP'.

    WHEN 'WERKS'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid,
                lcl_event_handler=>handle_hotspot      FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_mat
      it_fieldcatalog = gt_fcat.

ENDFORM.

*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM handle_double_click USING ps_row TYPE lvc_s_row
                               ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_mat INTO gs_mat INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.

      PERFORM get_mara USING gs_mat-matnr.

      IF gt_mara IS INITIAL.
        MESSAGE s000 WITH 'Material Information not found'.
        EXIT.
      ENDIF.

      CALL SCREEN '0101' STARTING AT 20 3.

  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM handle_hotspot USING ps_row TYPE lvc_s_row
                          ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_mat INTO gs_mat INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'WERKS'.
      PERFORM get_marc USING gs_mat-matnr
                             gs_mat-werks.

      IF gt_marc IS INITIAL.
        MESSAGE s000 WITH 'Plant Data for Material is not found'.
        EXIT.
      ENDIF.

      CALL SCREEN '0102' STARTING AT 20 3.

  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_mara USING pv_matnr TYPE mara-matnr.

  _clear gs_mara gt_mara.

  SELECT a~matnr a~mtart a~matkl
         b~maktx
         c~mtbez
         d~wgbez
    INTO CORRESPONDING FIELDS OF TABLE gt_mara
    FROM mara            AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr = b~matnr
     AND b~spras = sy-langu
    LEFT OUTER JOIN t134t AS c
      ON a~mtart = c~mtart
     AND c~spras = sy-langu
    LEFT OUTER JOIN t023t AS d
      ON a~matkl = d~matkl
     AND d~spras = sy-langu
   WHERE a~matnr = pv_matnr.

ENDFORM.

*& Form get_marc
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_MAT_MATNR
*&      --> GS_MAT_WERKS
*&---------------------------------------------------------------------*
FORM get_marc  USING pv_matnr TYPE marc-matnr
                     pv_werks TYPE marc-werks.

  _clear gs_marc gt_marc.

  SELECT a~matnr a~werks a~dispo a~disls
         b~dsnam
         c~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_marc
    FROM marc       AS a
   INNER JOIN t024d AS b
      ON a~dispo = b~dispo
     AND a~werks = b~werks
    LEFT OUTER JOIN makt AS c
      ON a~matnr = c~matnr
     AND c~spras = sy-langu
   WHERE a~matnr = pv_matnr
     AND a~werks = pv_werks.

ENDFORM.
