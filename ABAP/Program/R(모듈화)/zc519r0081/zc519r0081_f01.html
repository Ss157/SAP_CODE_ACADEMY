FORM get_purchase .

  REFRESH gt_purchase.

  SELECT ebeln, ebelp, bukrs, bsart, batxt,
         matnr, txz01, menge, meins
    INTO CORRESPONDING FIELDS OF TABLE @gt_purchase

*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat_1 USING pv_key
                       pv_field
                       pv_text
                       pv_ref_table
                       pv_ref_field.

  gs_fcat = VALUE #(
                     key       = pv_key
                     fieldname = pv_field
                     coltext   = pv_text
                     ref_table = pv_ref_table
                     ref_field = pv_ref_field
                   ).

  CASE pv_field.
    WHEN 'MENGE'.
      gs_fcat = VALUE #( BASE gs_fcat
                         qfieldname = 'MEINS'
                       ).

  ENDCASE.

  APPEND gs_fcat TO gt_fcat_1.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv_1 .

  CREATE OBJECT gcl_container_1
    EXPORTING
      container_name = 'GCL_CONTAINER_1'.

  CREATE OBJECT gcl_grid_1
    EXPORTING
      i_parent = gcl_container_1.

  gs_variant-report = sy-repid.
  gs_variant-handle = '1ST'.

  CALL METHOD gcl_grid_1->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_purchase
      it_fieldcatalog = gt_fcat_1.

ENDFORM.
