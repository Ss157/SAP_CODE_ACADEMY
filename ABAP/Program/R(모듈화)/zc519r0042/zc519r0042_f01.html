FORM get_airline .

  _clear gs_airline gt_airline.

  SELECT a~carrid a~carrname a~url
         b~connid b~fldate   b~planetype b~price b~currency
    INTO CORRESPONDING FIELDS OF TABLE gt_airline
    FROM scarr        AS a
   INNER JOIN sflight AS b
      ON a~carrid = b~carrid
   WHERE a~carrid IN so_carr
     AND b~connid IN so_conn.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

                lcl_event_handler=>handle_hotspot      FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_airline
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM handle_hotspot USING ps_row TYPE lvc_s_row
                          ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_airline INTO gs_airline INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'PLANETYPE'.
      PERFORM get_saplane USING gs_airline-planetype.

      IF gt_saplane IS INITIAL.
        MESSAGE s000 WITH 'Planetype Information is not found'.
        EXIT.
      ENDIF.

      CALL SCREEN '0101' STARTING AT 20 3.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM handle_double_click USING ps_row TYPE lvc_s_row
                               ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_airline INTO gs_airline INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'PLANETYPE'.
      EXIT.

    WHEN OTHERS.
      PERFORM get_sbook USING gs_airline-carrid
                              gs_airline-connid
                              gs_airline-fldate.

      IF gt_sbook IS INITIAL.
        MESSAGE s000 WITH 'Book Information is not found'.
        EXIT.
      ENDIF.

      CALL SCREEN '0102' STARTING AT 20 3.

  ENDCASE.

ENDFORM.

FORM get_saplane USING pv_planetype TYPE saplane-planetype.

  _clear : gs_saplane gt_saplane.

  SELECT planetype seatsmax tankcap
         cap_unit  weight   wei_unit
    INTO CORRESPONDING FIELDS OF TABLE gt_saplane
    FROM saplane
   WHERE planetype = pv_planetype.

ENDFORM.

FORM get_sbook USING pv_carrid TYPE sbook-carrid
                     pv_connid TYPE sbook-connid
                     pv_fldate TYPE sbook-fldate.

  _clear gs_sbook gt_sbook.

  SELECT carrid   connid     fldate bookid   customid
         custtype luggweight wunit  loccuram loccurkey
    INTO CORRESPONDING FIELDS OF TABLE gt_sbook
    FROM sbook
   WHERE carrid = pv_carrid
     AND connid = pv_connid
     AND fldate = pv_fldate.

ENDFORM.
