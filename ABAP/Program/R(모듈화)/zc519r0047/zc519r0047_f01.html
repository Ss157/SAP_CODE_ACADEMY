FORM get_company .

  _clear  gs_company gt_company.
  REFRESH gt_company_del.

  SELECT matnr werks ekgrp dispo beskz
    INTO CORRESPONDING FIELDS OF TABLE gt_company
    FROM zc519t0035
   WHERE werks  = pa_werks
     AND matnr IN so_matnr
     AND ekgrp IN so_ekgrp.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR  lv_tabix.

  LOOP AT gt_company INTO gs_company.

    lv_tabix = sy-tabix.

    CLEAR   ls_style.
    REFRESH gs_company-style.

    ls_style-fieldname = 'MATNR'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_company-style.
    CLEAR  ls_style.

    ls_style-fieldname = 'WERKS'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_company-style.
    CLEAR  ls_style.

    MODIFY gt_company FROM gs_company INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_company
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_company INTO gs_company INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_company TO gt_company_del.
      CLEAR  gs_company.

      DELETE gt_company INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc519t0035,
         lt_save LIKE TABLE OF ls_save,
         lt_del  LIKE TABLE OF ls_save.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_company_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_company_del TO lt_del.
    " gt_company_del은 style이 있기 때문에
    " zc519t0035와 똑같은 거로 delete해야 하기 때문에
    " lt_del를 이용해서 delete 해야 한다.

    IF lt_del IS NOT INITIAL.

      DELETE zc519t0035 FROM TABLE lt_del.

    ENDIF.

  ENDIF.

  MOVE-CORRESPONDING gt_company TO lt_save.

  IF lt_save IS NOT INITIAL.

    MODIFY zc519t0035 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.

      REFRESH gt_company_del.

      MESSAGE s002.
    ENDIF.

  ENDIF.

ENDFORM.
