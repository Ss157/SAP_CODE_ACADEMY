FORM get_purcha .

  CLEAR   : gs_purcha.
  REFRESH : gt_purcha.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~menge b~meins
    INTO CORRESPONDING FIELDS OF TABLE gt_purcha
    FROM ekko      AS a
   INNER JOIN ekpo AS b
      ON a~ebeln = b~ebeln
   WHERE a~bukrs  = pa_bukrs
     AND a~ebeln IN so_ebeln.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM get_text .

  CLEAR   : gs_t161t, gs_makt.
  REFRESH : gt_t161t, gt_makt.

  SELECT bsart batxt bstyp
    INTO CORRESPONDING FIELDS OF TABLE gt_t161t
    FROM t161t
   WHERE spras = sy-langu.

  SELECT matnr maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_makt
    FROM makt
   WHERE spras = sy-langu.

ENDFORM.

*& Form move_purcha
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM move_purcha .

  CLEAR  gt_purcha2.

  LOOP AT gt_purcha INTO gs_purcha.

    MOVE-CORRESPONDING gs_purcha TO gs_purcha2.

    READ TABLE gt_t161t INTO gs_t161t WITH KEY bsart = gs_purcha2-bsart
                                               bstyp = gs_purcha2-bstyp.

    IF sy-subrc = 0.

      gs_purcha2-batxt = gs_t161t-batxt.

    ENDIF.

    READ TABLE gt_makt INTO gs_makt WITH KEY matnr = gs_purcha2-matnr.

    IF sy-subrc = 0.

      gs_purcha2-maktx = gs_makt-maktx.

    ENDIF.

    APPEND gs_purcha2 TO gt_purcha2.
    CLEAR  gs_purcha2.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'MENGE'.
      gs_fcat-qfieldname = 'MEINS'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_purcha2
      it_fieldcatalog = gt_fcat.

ENDFORM.
