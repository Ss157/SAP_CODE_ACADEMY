FORM display_split .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container
      rows    = 3
      columns = 1.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_up.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 2
      column    = 1
    RECEIVING
      container = gcl_container_mid.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 3
      column    = 1
    RECEIVING
      container = gcl_container_down.

  CREATE OBJECT gcl_grid_up
    EXPORTING
      i_parent = gcl_container_up.

  CREATE OBJECT gcl_grid_mid
    EXPORTING
      i_parent = gcl_container_mid.

  CREATE OBJECT gcl_grid_down
    EXPORTING
      i_parent = gcl_container_down.

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid_up,
                lcl_event_handler=>handle_hotspot      FOR gcl_grid_mid.

  gs_variant-report = sy-repid.
  gs_variant-handle = 'UP'.

  CALL METHOD gcl_grid_up->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_scarr
      it_fieldcatalog = gt_fcat_up.

  gs_variant-handle = 'MID'.

  CALL METHOD gcl_grid_mid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_spfli
      it_fieldcatalog = gt_fcat_mid.

  gs_variant-handle = 'DN'.

  CALL METHOD gcl_grid_down->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_sflight
      it_fieldcatalog = gt_fcat_down.

ENDFORM.

*& Form handle_double_click

*&      --> E_COLUMN
*&      --> E_ROW
*&---------------------------------------------------------------------*
FORM handle_double_click  USING ps_col TYPE lvc_s_col
                                ps_row TYPE lvc_s_row.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_scarr INTO gs_scarr INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  PERFORM get_schedule USING gs_scarr-carrid.

  IF gt_spfli IS INITIAL.
    "Schedule not found
    MESSAGE s000 WITH gs_scarr-carrid TEXT-f01.

  ENDIF.

  REFRESH gt_sflight.
  PERFORM refresh_grid.
  PERFORM refresh_grid_flight.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_schedule
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_SCARR_CARRID
*&---------------------------------------------------------------------*
FORM get_schedule  USING pv_carrid TYPE scarr-carrid.

  _clear gs_spfli   gt_spfli.

  SELECT carrid connid airpfrom airpto deptime arrtime
    INTO CORRESPONDING FIELDS OF TABLE gt_spfli
    FROM spfli
   WHERE carrid = pv_carrid.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_spfli INTO gs_spfli INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'CARRID'.

      PERFORM get_flight USING gs_spfli-carrid
                               gs_spfli-connid.

      IF gt_sflight IS INITIAL.
        MESSAGE s000 WITH gs_spfli-carrid gs_spfli-connid
        'flight not found'.
      ENDIF.

  ENDCASE.

  PERFORM refresh_grid_flight.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_flight
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_SPFLI_CARRID
*&      --> GS_SPFLI_CONNID
*&---------------------------------------------------------------------*
FORM get_flight  USING pv_carrid TYPE spfli-carrid
                       pv_connid TYPE spfli-connid.

  _clear gs_sflight gt_sflight.

  SELECT carrid connid fldate price currency
    INTO CORRESPONDING FIELDS OF TABLE gt_sflight
    FROM sflight
   WHERE carrid = pv_carrid
     AND connid = pv_connid.

ENDFORM.
