FORM get_team .

  _clear  gs_team gt_team.
  REFRESH gt_team_del.

  SELECT team  team_name area  win   erdat
         erzet ernam     aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_team
    FROM zc519t0012
   WHERE team IN so_team.

ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  CLEAR  lv_tabix.

  LOOP AT gt_team INTO gs_team.

    lv_tabix = sy-tabix.

    _clear ls_style gs_team-style.

    ls_style-fieldname = 'TEAM'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_team-style.
    CLEAR  ls_style.

    MODIFY gt_team FROM gs_team INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_team
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH TEXT-f01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_team INTO gs_team INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_team TO gt_team_del.
      CLEAR  gs_team.

      DELETE gt_team INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc519t0012,
         lt_save LIKE TABLE OF ls_save,
         lt_del  LIKE TABLE OF ls_save.

  _clear  ls_save lt_save.
  REFRESH lt_del.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_team_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_team_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE zc519t0012 FROM TABLE lt_del.

    ENDIF.

  ENDIF.

  LOOP AT gt_team INTO gs_team.

    MOVE-CORRESPONDING gs_team TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY zc519t0012 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.
      REFRESH gt_team_del.

      MESSAGE s002.
    ENDIF.

  ENDIF.

ENDFORM.
