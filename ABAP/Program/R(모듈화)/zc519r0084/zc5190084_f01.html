FORM get_flight .

  _clear gs_sflight gt_sflight.

  SELECT carrid, connid, fldate, paymentsum, currency, planetype
    INTO CORRESPONDING FIELDS OF TABLE @gt_sflight
    FROM sflight
   WHERE carrid IN @so_carr
     AND connid IN @so_conn.

  IF sy-subrc <> 0.

    MESSAGE s001.
    STOP.

  ENDIF.

ENDFORM.

FORM set_fcat .

  gs_layout = VALUE #(
                       zebra      = 'X'
                       cwidth_opt = 'A'
                       sel_mode   = 'D'
                     ).

  PERFORM make_fcat USING :
  'X'  'CARRID'      ' '  'SFLIGHT'  ' ',
  'X'  'CONNID'      ' '  'SFLIGHT'  ' ',
  'X'  'FLDATE'      ' '  'SFLIGHT'  ' ',
  ' '  'PAYMENTSUM'  ' '  'SFLIGHT'  ' ',
  ' '  'CURRENCY'    ' '  'SFLIGHT'  ' ',
  ' '  'PLANETYPE'   ' '  'SFLIGHT'  ' '.

ENDFORM.

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat = VALUE #(
                      key       = pv_key
                      fieldname = pv_field
                      coltext   = pv_text
                      ref_table = pv_ref_table
                      ref_field = pv_ref_field
                   ).

*  CASE pv_field.
*    WHEN 'PAYMENTSUM'.
*      gs_fcat-cfieldname = 'CURRENCY'.
*
*  ENDCASE.

  APPEND VALUE #(
                  BASE gs_fcat
                  fieldname  = pv_field
                  cfieldname = COND #( WHEN pv_field = 'PAYMENTSUM' THEN 'CURRENCY' )
                ) TO gt_fcat.

*  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
