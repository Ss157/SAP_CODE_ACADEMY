FORM get_mat .

  _clear gs_mat gt_mat.

  SELECT matnr auart vkorg spart
         erdat erzet ernam
         aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM zc519t0022
   WHERE matnr IN so_matnr
     AND auart =  pa_auart.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.

    MESSAGE s000 WITH TEXT-f03 DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_mat INTO gs_mat INDEX gs_row-index.

    IF sy-subrc = 0.

      APPEND gs_mat TO gt_mat_del.
      CLEAR  gs_mat.

      DELETE gt_mat INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE zc519t0022,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear  ls_save lt_save.
  CLEAR   lv_commit.
  REFRESH lt_del.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_mat_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_mat_del TO lt_del.

    IF lt_del IS NOT INITIAL.

      DELETE zc519t0022 FROM TABLE lt_del.

      IF sy-dbcnt > 0.

        lv_commit = 'X'.

      ENDIF.

    ENDIF.

  ENDIF.

  LOOP AT gt_mat INTO gs_mat WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_mat TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY zc519t0022 FROM TABLE lt_save.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.
    REFRESH gt_mat_del.

    PERFORM reset_mark.

    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH TEXT-f04 DISPLAY LIKE 'W'.

  ENDIF.

ENDFORM.

FORM handle_data_changed  USING pv_modified
                                pt_modi     TYPE lvc_t_modi.

  CHECK pv_modified IS NOT INITIAL.

  LOOP AT pt_modi INTO DATA(ls_modi).

    READ TABLE gt_mat INTO gs_mat INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_mat-mark = 'X'.

    MODIFY gt_mat FROM gs_mat INDEX ls_modi-row_id TRANSPORTING mark.

  ENDLOOP.

ENDFORM.
