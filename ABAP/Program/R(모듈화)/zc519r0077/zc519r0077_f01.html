FORM modify_init .

  LOOP AT SCREEN.

    CASE 'X'.
      WHEN pa_rb1.
        CASE screen-group1.
          WHEN 'GP2'.
            screen-active = 0.

            MODIFY SCREEN.

        ENDCASE.

      WHEN pa_rb2.
        CASE screen-group1.
          WHEN 'GP1'.
            screen-active = 0.

            MODIFY SCREEN.

        ENDCASE.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM get_purchase .

FORM get_product .

  RANGES : lr_aufnr FOR aufk-aufnr.

  CLEAR   lr_aufnr.
  REFRESH gt_product.

  IF pa_aufnr IS NOT INITIAL.

    lr_aufnr-sign   = 'I'.
    lr_aufnr-option = 'EQ'.
    lr_aufnr-low    = pa_aufnr.

    APPEND lr_aufnr.

  ENDIF.

  SELECT aufnr, posnr, auart, auart_txt, bukrs, werks,
         matnr, gstrp, gltrp, rsnum,     psmng, amein
    INTO CORRESPONDING FIELDS OF TABLE @gt_product
    FROM ZC519cds0011
   WHERE aufnr IN @lr_aufnr.

  IF sy-subrc <> 0.

    MESSAGE s001 DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.

FORM display_alv .

  FIELD-SYMBOLS : <lft_itab> TYPE ANY TABLE.

  UNASSIGN <lft_itab>.

  " 인터널 테이블은 gt_purchase, gt_product 중에
  " 하나이기 때문에 필드 심볼을 사용해서 set_table...을
  " 한 번하도록 변경했다.
  CASE 'X'.
    WHEN pa_rb1.
      ASSIGN gt_purchase TO <lft_itab>.

      gs_variant-handle = 'RB1'.

    WHEN pa_rb2.
      ASSIGN gt_product TO <lft_itab>.

      gs_variant-handle = 'RB2'.

  ENDCASE.

  " 굳이 안 해도 됨. loop 돌릴 때 하기
  IF <lft_itab> IS NOT ASSIGNED.
    MESSAGE s000 WITH 'Table name is wrong'
    DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = <lft_itab>
      it_fieldcatalog = gt_fcat.

ENDFORM.
