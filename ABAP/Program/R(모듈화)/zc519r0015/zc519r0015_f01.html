FORM get_account .

  CLEAR   : gs_account.
  REFRESH : gt_account.

  SELECT a~bukrs a~gjahr a~belnr a~blart a~budat
         b~buzei b~bschl b~koart b~shkzg b~fcsl
         c~ltext
         d~ltext AS bschl_txt
    INTO CORRESPONDING FIELDS OF TABLE gt_account
    FROM bkpf      AS a
   INNER JOIN bseg AS b
      ON a~bukrs = b~bukrs
     AND a~belnr = b~belnr
     AND a~gjahr = b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart = c~blart
     AND c~spras = sy-langu
    LEFT OUTER JOIN tbslt AS d
      ON b~bschl = d~bschl
     AND d~spras = sy-langu
   WHERE a~bukrs IN so_bukrs
     AND a~gjahr  = pa_gjahr.

  IF sy-subrc <> 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM modify_account .

  DATA : lv_tabix TYPE sy-tabix.
  CLEAR  lv_tabix.

  LOOP AT gt_account INTO gs_account.

    lv_tabix = sy-tabix.

    CASE gs_account-shkzg.
      WHEN 'H'.
        gs_account-fcsl *= -1.

    ENDCASE.

    gs_account-rfccur = 'KRW'.

    MODIFY gt_account FROM gs_account INDEX lv_tabix
    TRANSPORTING fcsl rfccur.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'FCSL'.
      gs_fcat-cfieldname = 'RFCCUR'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_account
      it_fieldcatalog = gt_fcat.

ENDFORM.
