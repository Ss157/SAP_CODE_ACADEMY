FORM get_airline .

  REFRESH : gt_scarr.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr

FORM display_split .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

* 상위 컨테이너를 나눈다.
* 나눔과 동시에 내부에 컨테이너가 나눈만큼 생김
* 즉 create object가 생긴다는 것임.
  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container "무슨 컨테이너를 나눌 것인가?
      rows    = 2             "행
      columns = 1.            "열

* 1행 1열의 컨테이너 받아오기
  " 메소드에서는 return이니 호출부에서는 receiving을 사용한다.
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 2
      column    = 1
    RECEIVING
      container = gcl_container_tmp.

  CREATE OBJECT gcl_splitter2
    EXPORTING
      parent  = gcl_container_tmp
      rows    = 1
      columns = 2.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_top.

* 2행 1열의 컨테이너 받아오기
  CALL METHOD gcl_splitter2->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_mid.

  CALL METHOD gcl_splitter2->get_container
    EXPORTING
      row       = 1
      column    = 2
    RECEIVING
      container = gcl_container_bot.

* 1행 1열 위치에 Grid 붙이기
  CREATE OBJECT gcl_grid_top
    EXPORTING
      i_parent = gcl_container_top.

* 2행 1열 위치에 Grid 붙이기
  CREATE OBJECT gcl_grid_mid
    EXPORTING
      i_parent = gcl_container_mid.

  CREATE OBJECT gcl_grid_bot
    EXPORTING
      i_parent = gcl_container_bot.

*  CALL METHOD gcl_splitter->set_row_height
*    EXPORTING
*      id     = 1 "몇 번째 행
*      height = 30.

  gs_variant-report = sy-repid. "현재 이 프로그램
  gs_variant-handle = 'TOP'. "handle에 각 alv에 구분자를 주기
  "handle = alv 별로 variant를 구분해서 주기 위해서 쓴다.
  " TOP 부분에 의미있는 이름을 주지 말아야 함.

  CALL METHOD gcl_grid_top->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_scarr
      it_fieldcatalog = gt_fcat_top.

  gs_variant-handle = 'MID'. "handle에 각 alv에 구분자를 주기
  "handle = alv 별로 variant를 구분해서 주기 위해서 쓴다.

  CALL METHOD gcl_grid_mid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_spfli
      it_fieldcatalog = gt_fcat_mid.

  gs_variant-handle = 'BOT'. "handle에 각 alv에 구분자를 주기
  "handle = alv 별로 variant를 구분해서 주기 위해서 쓴다.

  CALL METHOD gcl_grid_bot->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_sflight
      it_fieldcatalog = gt_fcat_bot.

ENDFORM.
