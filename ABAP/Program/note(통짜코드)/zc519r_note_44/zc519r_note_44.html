         carrid     TYPE sflight-carrid,
         carrname   TYPE zc519v0032-carrname,
         connid     TYPE sflight-connid,
         fldate     TYPE sflight-fldate,
         cityfrom   TYPE zc519v0032-cityfrom,
         cityto     TYPE zc519v0032-cityto,
         planetype  TYPE sflight-planetype,
         seatsmax_b TYPE sflight-seatsmax_b,

         fldate    TYPE sflight-fldate,
         cityfrom  TYPE zc519v0032-cityfrom,
         cityto    TYPE zc519v0032-cityto,

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  PARAMETERS     : pa_carr TYPE sflight-carrid OBLIGATORY DEFAULT 'LH'.
  SELECT-OPTIONS : so_conn FOR  sflight-connid OBLIGATORY.

SELECT a~carrid   a~connid   a~fldate a~planetype a~seatsmax_b
       b~carrname b~cityfrom b~cityto b~url
  INTO CORRESPONDING FIELDS OF TABLE gt_itab1
  FROM sflight AS a
 INNER JOIN zc519v0032 AS b
    ON a~carrid = b~carrid
   AND a~connid = b~connid
 WHERE a~carrid  = pa_carr
   AND a~connid IN so_conn.

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

cl_demo_output=>display( gt_itab1 ).

LOOP AT gt_itab1 INTO gs_itab1.

  MOVE-CORRESPONDING gs_itab1 TO gs_itab2.

  CASE gs_itab1-seatsmax_b.
    WHEN '30'.
      gs_itab2-planetype = '747-400'.

    WHEN '21'.
      gs_itab2-planetype = 'A340-600'.

    WHEN OTHERS.
      CLEAR gs_itab2-planetype.

  ENDCASE.

  APPEND gs_itab2 TO gt_itab2.
  CLEAR : gs_itab1, gs_itab2.
ENDLOOP.

cl_demo_output=>display( gt_itab2 ).
