* select로 6개의 테이블을 조인하고 있다.
* 여기서 text table과 조인하기 때문에 언어키가 중요하다.
IF pa_rb3 = 'X'.
  SELECT a~matnr a~werks  a~stlan
         b~stlty b~stlal  b~stkoz b~andat b~stlnr b~bmein
         c~postp c~stlkn  c~stpoz c~idnrk c~menge c~posnr
         d~maktx
         e~name1
         f~ptext
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mast AS a
   INNER JOIN stko AS b
      ON a~stlnr = b~stlnr
   INNER JOIN stpo AS c
      ON a~stlnr = c~stlnr
   INNER JOIN makt AS d
      ON a~matnr = d~matnr
     AND d~spras = sy-langu
   INNER JOIN t001w AS e
      ON a~werks = e~werks
*   AND e~spras = sy-langu
   INNER JOIN t418t AS f
      ON c~postp  = f~postp
     AND f~spras  = sy-langu
   WHERE a~werks  = pa_werks
     AND a~matnr IN so_matnr
     AND b~stlnr IN so_stlnr.
ELSE.

  SELECT a~matnr a~werks  a~stlan
         b~stlty b~stlal  b~stkoz b~andat b~stlnr b~bmein
         c~postp c~stlkn  c~stpoz c~idnrk c~menge c~posnr
         d~maktx
         e~name1
         f~ptext
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mast AS a
   INNER JOIN stko AS b
      ON a~stlnr = b~stlnr
   INNER JOIN stpo AS c
      ON a~stlnr = c~stlnr
   INNER JOIN makt AS d
      ON a~matnr = d~matnr
     AND d~spras = sy-langu
   INNER JOIN t001w AS e
      ON a~werks = e~werks
*   AND e~spras = sy-langu
   INNER JOIN t418t AS f
      ON c~postp  = f~postp
     AND f~spras  = sy-langu
   WHERE a~werks  = pa_werks
     AND a~matnr IN so_matnr
     AND b~stlnr IN so_stlnr
     AND c~postp  = gv_rdchk.
ENDIF.

IF sy-subrc <> 0.
