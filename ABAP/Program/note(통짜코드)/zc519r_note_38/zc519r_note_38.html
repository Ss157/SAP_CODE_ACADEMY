REPORT zc519r_note_38 MESSAGE-ID zc519m.

TABLES : mara, marc.

DATA : BEGIN OF gt_data OCCURS 0,
         matnr TYPE mara-matnr,
         maktx TYPE makt-maktx, "matnr의 내역
         werks TYPE marc-werks,
         lgort TYPE mard-lgort,
         mtart TYPE mara-mtart,
         mtbez TYPE t134t-mtbez, "mtart의 내역
         matkl TYPE mara-matkl,
         wgbez TYPE t023t-wgbez, "matkl의 내역
         pstat TYPE marc-pstat,
         ekgrp TYPE marc-ekgrp,
         eknam TYPE t024-eknam, "ekgrp의 내역 value table
         labst TYPE mard-labst,
         meins TYPE mara-meins,
         dispo TYPE marc-dispo,
       END OF gt_data,

       gt_t024  TYPE TABLE OF t024 WITH HEADER LINE,

       lv_tabix TYPE sy-tabix.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  SELECT-OPTIONS : so_matnr FOR mara-matnr NO INTERVALS,
                   so_mtart FOR mara-mtart NO-EXTENSION,
                   so_werks FOR marc-werks NO INTERVALS NO-EXTENSION.

SELECTION-SCREEN END OF BLOCK bl1.

CLEAR   : gt_data, gt_t024.
REFRESH : gt_data, gt_t024.

SELECT a~matnr a~mtart a~matkl a~meins
       b~werks b~pstat b~ekgrp b~dispo
       c~lgort c~labst
       d~maktx e~mtbez f~wgbez
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM mara AS a
  INNER JOIN marc AS b
    ON a~matnr = b~matnr
  LEFT OUTER JOIN mard AS c
    ON b~matnr = c~matnr
   AND b~werks = c~werks
  LEFT OUTER JOIN makt AS d
    ON a~matnr = d~matnr
   AND d~spras = sy-langu
  LEFT OUTER JOIN t134t AS e
    ON e~spras = sy-langu
   AND a~mtart = e~mtart
  LEFT OUTER JOIN t023t AS f
    ON f~spras  = sy-langu
   AND a~matkl  = f~matkl
 WHERE a~matnr IN so_matnr
   AND a~mtart IN so_mtart
   AND b~werks IN so_werks.

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

SELECT ekgrp eknam
  INTO CORRESPONDING FIELDS OF TABLE gt_t024
  FROM t024.

*cl_demo_output=>display( gt_data[] ).

SORT gt_t024 BY ekgrp.

LOOP AT gt_data.

  lv_tabix = sy-tabix.

  READ TABLE gt_t024 WITH KEY ekgrp = gt_data-ekgrp
                     BINARY SEARCH.

  IF sy-subrc = 0.
    gt_data-eknam = gt_t024-eknam.

    MODIFY gt_data INDEX lv_tabix TRANSPORTING eknam.

  ENDIF.

ENDLOOP.

cl_demo_output=>display( gt_data[] ).
