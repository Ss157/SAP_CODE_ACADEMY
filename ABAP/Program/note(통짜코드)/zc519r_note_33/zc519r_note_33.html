REPORT zc519r_note_33 MESSAGE-ID zc519m.

DATA : BEGIN OF ls_mat,
         matnr TYPE zc519t0033-matnr,
         werks TYPE zc519t0033-werks,
         mtart TYPE zc519t0032-mtart,
         pstat TYPE zc519t0032-pstat,
         ekgrp TYPE zc519t0033-ekgrp,
         dispo TYPE zc519t0033-dispo,
         beskz TYPE zc519t0033-beskz,
       END OF ls_mat,
       lt_mat   LIKE TABLE OF ls_mat,

       lv_tabix TYPE sy-tabix.

SELECT matnr werks mtart pstat
  INTO CORRESPONDING FIELDS OF TABLE lt_mat
  FROM zc519t0032.

SELECT matnr werks ekgrp dispo beskz
  APPENDING CORRESPONDING FIELDS OF TABLE lt_mat
  FROM zc519t0033.

* 이 코드에서는 두 테이블이 모두 주 테이블이기 때문에
* 둘 다 안 나올 때
IF lt_mat IS INITIAL.
  MESSAGE s001.
  STOP.
ENDIF.

cl_demo_output=>display( lt_mat ).

LOOP AT lt_mat INTO ls_mat.

  lv_tabix = sy-tabix.

  ls_mat-ekgrp = 'Z05'.
  ls_mat-pstat = 'DAB'.

  MODIFY lt_mat FROM ls_mat INDEX lv_tabix
  TRANSPORTING ekgrp pstat.

ENDLOOP.

cl_demo_output=>display( lt_mat ).
