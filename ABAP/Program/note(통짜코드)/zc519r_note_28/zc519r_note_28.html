* bkpf, bseg를 조인하여 ltext, koart_text, shkzg_text를 제외한 필드에
* 값을 채운다.
SELECT a~bukrs a~belnr a~gjahr a~blart a~budat a~bldat
       b~buzei b~bschl b~koart b~shkzg b~dmbtr
       c~ltext
  INTO CORRESPONDING FIELDS OF TABLE lt_account
  FROM bkpf AS a
 INNER JOIN bseg AS b
    ON a~bukrs = b~bukrs
   AND a~belnr = b~belnr
   AND a~gjahr = b~gjahr
  LEFT OUTER JOIN t003t AS c
    ON a~blart = c~blart
   AND c~spras = sy-langu
 WHERE a~gjahr = pa_gjahr.
