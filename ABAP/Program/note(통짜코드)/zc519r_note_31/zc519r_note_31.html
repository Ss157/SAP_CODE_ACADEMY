REPORT zc519r_note_31 MESSAGE-ID zc519m.

DATA : ls_sflight TYPE sflight,
       lt_sflight LIKE TABLE OF ls_sflight,

       BEGIN OF ls_airline,
         carrid   TYPE scarr-carrid,
         connid   TYPE sflight-connid,
         fldate   TYPE sflight-fldate,
         carrname TYPE scarr-carrname,
         price    TYPE sflight-price,
         currency TYPE sflight-currency,
         url      TYPE scarr-url,
       END OF ls_airline,
       lt_airline LIKE TABLE OF ls_airline,

       lv_tabix   TYPE sy-tabix.

*CLEAR   :
*refresh :

SELECT carrid   connid     fldate     price      currency   planetype  seatsmax
       seatsocc paymentsum seatsmax_b seatsocc_b seatsmax_f seatsocc_f
  INTO CORRESPONDING FIELDS OF TABLE lt_sflight
  FROM sflight
 WHERE carrid IN ('LH', 'DL').

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

MOVE-CORRESPONDING lt_sflight TO lt_airline.

LOOP AT lt_airline INTO ls_airline.

  lv_tabix = sy-tabix.

  CASE ls_airline-carrid.
    WHEN 'LH'.
      ls_airline-currency = '유로'.

      MODIFY lt_airline FROM ls_Airline INDEX lv_tabix
      TRANSPORTING currency.
  ENDCASE.

ENDLOOP.

cl_demo_output=>display( lt_airline ).
