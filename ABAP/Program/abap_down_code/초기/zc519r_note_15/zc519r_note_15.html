REPORT zc519r_note_15 MESSAGE-ID zc519m.

* TABLES는 select option을 사용할 때, 사용되는 것으로,
* DATA : zc519t0029 TYPE zc519t0029.과 의미가 같다.
* DATA : zc519t0030 TYPE zc519t0030.과 의미가 같다.
TABLES : zc519t0029, zc519t0030.

DATA : BEGIN OF gs_idol,
         group_id        TYPE zc519t0029-group_id,
         group_txt       TYPE zc519t0029-group_txt,
         member_id       TYPE zc519t0030-member_id,
         member_name     TYPE zc519t0030-member_name,
         member_gender   TYPE zc519t0030-member_gender,
         member_addr     TYPE zc519t0030-member_addr,
         member_phone    TYPE zc519t0030-member_phone,
         group_url       TYPE zc519t0029-group_url,
         gender_text(10),
       END OF gs_idol,

       gt_idol  LIKE TABLE OF gs_idol,

       lv_tabix TYPE sy-tabix.

* so_groid는 select options이지만 NO INTERVALS, NO-EXTENSION를 이용해서
* 단일값만 받을 수 있게 되었다.
* pa_memid는 파라미터인데 파라미터는 빈값을 넣었을 때,
* 그 값을 처리해서 빈 테이블이 출력된다.
* 따라서 원래 파라미터는 mandatory이고 default로 값을 지정해 주면 좋으나
* 명세에 mandatory하지 말라고 해서 그 것은 하지 않았다.
* 빈 테이블이 출력되는 것을 방지하기 위해서는
* select options에 NO INTERVALS, NO-EXTENSION를 이용하는 방법도 있다
SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME  TITLE TEXT-t01.

  SELECT-OPTIONS : so_groid FOR zc519t0029-group_id  NO INTERVALS
                                                     NO-EXTENSION.

  PARAMETERS     : pa_memid TYPE zc519t0030-member_id DEFAULT '1'.

SELECTION-SCREEN END OF BLOCK bl1.

CLEAR   : gs_idol.
REFRESH : gt_idol.

* zc519t0029, zc519t0030 두 테이블을 조인하여 필드를 select했다.
* 위에서 만든 so_groid, pa_memid 또한 where 조건에서 실행했다.
SELECT a~group_id    a~group_txt    a~group_url
       b~member_id   b~member_name  b~member_gender
       b~member_addr b~member_phone
  INTO CORRESPONDING FIELDS OF TABLE gt_idol
  FROM zc519t0029 AS a
 INNER JOIN zc519t0030 AS b
    ON a~group_id  = b~group_id
 WHERE a~group_id IN so_groid
   AND b~member_id = pa_memid.

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

* loop 문에서 member_gender가 M인지 F인지에 따라서
* gender_text에 남자, 여자라고 값을 대입한다.
LOOP AT gt_idol INTO gs_idol.

  lv_tabix = sy-tabix.

  CASE gs_idol-member_gender.
    WHEN 'M'.
      gs_idol-gender_text = '남자'.

      MODIFY gt_idol FROM gs_idol INDEX lv_tabix
        TRANSPORTING gender_text.

    WHEN 'F'.
      gs_idol-gender_text = '여자'.

      MODIFY gt_idol FROM gs_idol INDEX lv_tabix
        TRANSPORTING gender_text.

  ENDCASE.

ENDLOOP.

cl_demo_output=>display( gt_idol ).
