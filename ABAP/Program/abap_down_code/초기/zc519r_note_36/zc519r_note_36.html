REPORT zc519r_note_36 MESSAGE-ID zc519m.

DATA : gs_scarr   TYPE scarr,
       gt_scarr   LIKE TABLE OF gs_scarr,
       gt_scarr_h LIKE TABLE OF gs_scarr WITH HEADER LINE,

       lv_tabix   TYPE sy-tabix.

* gt_scarr_h = header
* gt_scarr_h[] = body
CLEAR   : gs_scarr, gt_scarr, gt_scarr_h, gt_scarr_h[], lv_tabix.
* gt_scarr_h는 head이지만 refresh는 인터널테이블을 청소하는 것이기 때문에 body청소가 됨.
REFRESH : gt_scarr, gt_scarr_h.

SELECT carrid carrname currcode url
  INTO CORRESPONDING FIELDS OF TABLE gt_scarr_h
  FROM scarr.

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

* gt_scarr_h에 있는 레코드에서 carrid가 'JL'이면
* 'KA'로 바꾸고 carrname을 'Korean Air'로 바꿀 것

LOOP AT gt_scarr_h.

  lv_tabix = sy-tabix.

  CASE gt_scarr_h-carrid.
    WHEN 'JL'.
      gt_scarr_h-carrid   = 'KA'.
      gt_scarr_h-carrname = 'Korean Air'.

      MODIFY gt_scarr_h INDEX lv_tabix
      TRANSPORTING carrid carrname.

  ENDCASE.

ENDLOOP.

cl_demo_output=>display( gt_scarr_h[] ).
