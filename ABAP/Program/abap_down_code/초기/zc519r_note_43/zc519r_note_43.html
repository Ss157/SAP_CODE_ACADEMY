REPORT zc519r_note_43 MESSAGE-ID zc519m.

DATA : gs_data  TYPE LINE OF zc519tt002,
       gt_data  TYPE zc519tt002,

       gt_t005t TYPE TABLE OF t005t WITH HEADER LINE,
       gt_t009t TYPE TABLE OF t009t WITH HEADER LINE,

       lv_tabix TYPE sy-tabix.

SELECT bukrs butxt land1 periv kkber
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM t001.

IF sy-subrc <> 0.
  MESSAGE s001.
  STOP.
ENDIF.

SELECT land1 landx
  INTO CORRESPONDING FIELDS OF TABLE gt_t005t
  FROM t005t
 WHERE spras = sy-langu.

SELECT periv ltext
  INTO CORRESPONDING FIELDS OF TABLE gt_t009t
  FROM t009t
 WHERE spras = sy-langu.

SELECT kkber kkbtx
  INTO CORRESPONDING FIELDS OF TABLE gt_t014t
  FROM t014t
 WHERE spras = sy-langu.

LOOP AT gt_data INTO gs_data.

  lv_tabix = sy-tabix.

  READ TABLE gt_t005t WITH KEY land1 = gs_data-land1.

  IF sy-subrc = 0.
    gs_data-landx = gt_t005t-landx.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING landx.
  ENDIF.

  READ TABLE gt_t009t WITH KEY periv = gs_data-periv.

  IF sy-subrc = 0.
    gs_data-ltext = gt_t009t-ltext.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING ltext.
  ENDIF.

  READ TABLE gt_t014t WITH KEY kkber = gs_data-kkber.

  IF sy-subrc = 0.
    gs_data-kkbtx = gt_t014t-kkbtx.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING kkbtx.
  ENDIF.

ENDLOOP.
