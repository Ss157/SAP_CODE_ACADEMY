SELECT carrid   connid fldate  bookid  customid
       custtype smoker invoice
  INTO CORRESPONDING FIELDS OF TABLE lt_sbook
  FROM sbook
 WHERE carrid   = 'LH'
   AND connid   IN ( '0401', '2402' )
   AND customid >= '4387'.

IF sy-subrc <> 0.

  MESSAGE s001.
  STOP.

ENDIF.

LOOP AT lt_sbook INTO ls_sbook.

  lv_tabix = sy-tabix.

  CASE 'X'.
    WHEN ls_sbook-invoice.
      ls_sbook-billing = 'X'.

      MODIFY lt_sbook FROM ls_sbook INDEX lv_tabix
      TRANSPORTING billing.

    WHEN ls_sbook-smoker.
      ls_sbook-billing = 'X'.

      MODIFY lt_sbook FROM ls_sbook INDEX lv_tabix
      TRANSPORTING billing.

  ENDCASE.

ENDLOOP.
