REPORT zc519r_note_41 MESSAGE-ID zc519m.

TABLES : sflight, sbook.

DATA : BEGIN OF ls_sflight1,
         carrid    TYPE sflight-carrid,
         connid    TYPE sflight-connid,
         fldate    TYPE sflight-fldate,
         planetype TYPE sflight-planetype,
         currency  TYPE sflight-currency,
         bookid    TYPE sbook-bookid,
         customid  TYPE sbook-customid,
         custtype  TYPE sbook-custtype,
         class     TYPE sbook-class,
         agencynum TYPE sbook-agencynum,
       END OF ls_sflight1,
       lt_sflight1 LIKE TABLE OF ls_sflight1,

       BEGIN OF ls_sflight2,
         carrid    TYPE sflight-carrid,
         connid    TYPE sflight-connid,
         fldate    TYPE sflight-fldate,
         bookid    TYPE sbook-bookid,
         customid  TYPE sbook-customid,
         custtype  TYPE sbook-custtype,
         agencynum TYPE sbook-agencynum,
       END OF ls_sflight2,
       lt_sflight2 LIKE TABLE OF ls_sflight2.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  PARAMETERS     : pa_carr TYPE sflight-carrid OBLIGATORY.
  SELECT-OPTIONS : so_conn FOR sflight-connid OBLIGATORY.

  PARAMETERS     : pa_type TYPE sflight-planetype AS LISTBOX
                                                  VISIBLE LENGTH 10
                                                  OBLIGATORY.
  SELECT-OPTIONS : so_bkid FOR sbook-bookid.

SELECTION-SCREEN END OF BLOCK bl1.

CLEAR   : ls_sflight1, ls_sflight2.
REFRESH : lt_sflight1, lt_sflight2.

SELECT a~carrid a~connid   a~fldate   a~planetype a~currency
       b~bookid b~customid b~custtype b~class     b~agencynum
  INTO CORRESPONDING FIELDS OF TABLE lt_sflight1
  FROM sflight AS a
 INNER JOIN sbook AS b
    ON a~carrid    = b~carrid
   AND a~connid    = b~connid
   AND a~fldate    = b~fldate
 WHERE a~carrid    = pa_carr
   AND a~planetype = pa_type
   AND a~connid   IN so_conn
   AND b~bookid   IN so_bkid.

IF sy-subrc <> 0.

  MESSAGE s001.
  STOP.

ENDIF.

cl_demo_output=>display( lt_sflight1 ).

LOOP AT lt_sflight1 INTO ls_sflight1.

  IF ls_sflight1-custtype = 'B'.
    MOVE-CORRESPONDING ls_sflight1 TO ls_sflight2.

    APPEND ls_sflight2 TO lt_sflight2.
    CLEAR  ls_sflight2.
  ENDIF.

ENDLOOP.

cl_demo_output=>display( lt_sflight2 ).
