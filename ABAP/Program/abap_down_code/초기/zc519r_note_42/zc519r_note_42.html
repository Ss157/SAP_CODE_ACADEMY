       BEGIN OF gt_itab2 OCCURS 0,

         matkl TYPE mara-matkl,
         wgbez TYPE t023t-wgbez, "matkl의 내역

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  SELECT-OPTIONS : so_matnr FOR mara-matnr.
  PARAMETERS     : pa_werks TYPE marc-werks OBLIGATORY DEFAULT '0001'.

SELECTION-SCREEN END OF BLOCK bl1.

CLEAR   : gt_itab1, gt_itab2.

SELECT a~matnr a~mtart a~matkl
       b~werks b~dispo b~ekgrp
  INTO CORRESPONDING FIELDS OF TABLE gt_itab1
  FROM mara AS a
 INNER JOIN marc AS b
    ON a~matnr = b~matnr
 WHERE a~matnr IN so_matnr
   AND b~werks  = pa_werks.

IF sy-subrc <> 0.

  MESSAGE s001.
  STOP.

ENDIF.

SELECT matnr maktx
  INTO CORRESPONDING FIELDS OF TABLE gt_makt
  FROM makt
 WHERE spras  = sy-langu
   AND matnr IN so_matnr.

SELECT matkl wgbez
  INTO CORRESPONDING FIELDS OF TABLE gt_t023t
  FROM t023t
 WHERE spras = sy-langu.

SELECT dispo dsnam
  INTO CORRESPONDING FIELDS OF TABLE gt_t024d
  FROM t024d
 WHERE werks = pa_werks. "여기 작성 필수

SELECT ekgrp eknam
  INTO CORRESPONDING FIELDS OF TABLE gt_t024
  FROM t024.

LOOP AT gt_itab1.

  MOVE-CORRESPONDING gt_itab1 TO gt_itab2.

  READ TABLE gt_makt WITH KEY matnr = gt_itab2-matnr.

  IF sy-subrc = 0.
    gt_itab2-maktx = gt_makt-maktx.

  ENDIF.

  READ TABLE gt_t023t WITH KEY matkl = gt_itab2-matkl.

  IF sy-subrc = 0.
    gt_itab2-wgbez = gt_t023t-wgbez.

  ENDIF.

  READ TABLE gt_t024d WITH KEY dispo = gt_itab2-dispo.

  IF sy-subrc = 0.
    gt_itab2-dsnam = gt_t024d-dsnam.

  ENDIF.

  READ TABLE gt_t024 WITH KEY ekgrp = gt_itab2-ekgrp.

  IF sy-subrc = 0.
    gt_itab2-eknam = gt_t024-eknam.

  ENDIF.

  APPEND gt_itab2.
  CLEAR  gt_itab2.

ENDLOOP.
