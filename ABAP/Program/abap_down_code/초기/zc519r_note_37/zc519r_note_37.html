SELECT ktopl ktplt
  INTO CORRESPONDING FIELDS OF TABLE gt_t004t
  FROM t004t
 WHERE spras = sy-langu
   AND ktopl = pa_ktopl.

SELECT saknr txt20
  INTO CORRESPONDING FIELDS OF TABLE gt_skat
  FROM skat
 WHERE spras = sy-langu
   AND ktopl = pa_ktopl.

SELECT ktoks txt30
  INTO CORRESPONDING FIELDS OF TABLE gt_t077z
  FROM t077z
 WHERE spras = sy-langu
   AND ktopl = pa_ktopl.

       gt_skat  BY saknr,
       gt_t077z BY ktoks.
